\begin{abstract}
For a prime $p\equiv1\pmod 4$ and a positive integer $a\equiv3\pmod4$ put
$x_a=(p+a)/4$.  The equation $4/p=1/x+1/y+1/z$ has a solution with
denominator $x=x_a$ exactly when $-1$ or $-p$ is a ratio $u/v$ modulo $a$
of coprime $u,v$ with $uv\mid x_a$.  The least such $a$, the \emph{window
statistic} $\amin(p)$, is finite for every $p$ if and only if the
Erd\H{o}s--Straus conjecture holds.  We study how often $\amin(p)$ is
large.  For every fixed $Z$ the number of primes $p\le N$ with
$\amin(p)>Z$ is $\ll_Z N/(\log N)^{1+J(Z)/2}$, where $J(Z)$ is the number of
$a\le Z$ with $a\equiv3\pmod4$; this is the exponent of the natural random
model.  The bound is attained for $Z=3$: at least $\gg x/(\log x)^{3/2}$
primes $p\le x$ in Mordell's hard classes have $\amin(p)\ge7$ (modulo
cited sieve theorems).  On the Elliott--Halberstam conjecture it is also
attained for $Z=7$.  Unconditionally, two windows run into the parity
problem: we exhibit two explicit sets of primes with the same sieve data
at every level allowed by Bombieri--Vinogradov, of which one has window~3
failing for $\gg x/(\log x)^{3/2}$ elements and the other for none.  A
model computation suggesting that Type~I information plus parity does not
suffice for two windows is recorded as a remark with its limited scope.
Nothing here proves the Erd\H{o}s--Straus conjecture.
\end{abstract}

\section{Introduction}\label{sec:intro}

The Erd\H{o}s--Straus conjecture asserts that for every integer $n\ge2$
\begin{equation}\label{eq:ES}
 \frac4n=\frac1x+\frac1y+\frac1z
\end{equation}
has a solution in positive integers.  It suffices to treat primes $n=p$,
and Mordell's polynomial identities \cite{Mordell} settle every prime
outside the six classes
\[
 p\equiv1,\ 121,\ 169,\ 289,\ 361,\ 529\pmod{840}.
\]
We call these primes \emph{Mordell-hard}.  Each of the six classes is a
square modulo $3$, $5$, $7$ and $8$, so Mordell-hard primes satisfy
$p\equiv1\pmod{24}$ and $\leg p3=\leg p5=\leg p7=1$.

\subsection{The window statistic}
Fix a prime $p\equiv1\pmod4$.  For an integer $x$ coprime to an odd modulus
$a$ let
\[
 \Rat_a(x)=\Bigl\{\textstyle\prod_{r\mid x}r^{f_r}\bmod a:\ |f_r|\le v_r(x)\Bigr\}
 =\{u v^{-1}\bmod a:\ u,v\ge1,\ \gcd(u,v)=1,\ uv\mid x\}\subseteq(\Z/a\Z)^\times
\]
(the product is over primes $r$; the two descriptions agree by putting the
positive exponents into $u$ and the negative ones into $v$).  For
$a\equiv3\pmod4$, $a\ge3$, put
\[
 x_a=\frac{p+a}4,
\]
an integer.  We say that the \emph{window} $a$ \emph{succeeds} if
$p\nmid x_a$ and $\Rat_a(x_a)\cap\{-1,-p\}\ne\varnothing$, and that it
\emph{fails} otherwise.  (For $a<3p$ we have $x_a<p$, so $p\nmid x_a$
automatically.)  Define
\[
 \amin(p)=\min\{a\equiv3\ (\mathrm{mod}\ 4):\ \text{window $a$ succeeds}\}
 \in\{3,7,11,\dots\}\cup\{\infty\}.
\]
Theorem~\ref{thm:frame} below shows that \eqref{eq:ES} is solvable for
$n=p$ if and only if $\amin(p)<\infty$; the window $a$ is the denominator
$x=x_a$ of a solution coprime to $p$, and the two targets $-1$, $-p$
correspond to the two classical solution types.  Thus the
Erd\H{o}s--Straus conjecture is the statement that the search
$a=3,7,11,\dots$ always terminates, and $\amin(p)$ is its length.
Numerically $\amin(p)$ is small: for every Mordell-hard $p<10^8$ one has
$\amin(p)\le107$ (Remark~\ref{rem:data}).

\subsection{Results}
Write $J(Z)=\#\{a\le Z:\ a\equiv3\pmod4\}=\lfloor(Z+1)/4\rfloor$.
In the natural random model (Remark~\ref{rem:model}) each window fails
with probability of order $(\log p)^{-1/2}$, roughly independently, so
one expects
\begin{equation}\label{eq:model-intro}
 \#\{p\le N:\ \amin(p)>Z\}\approx \pi(N)\,(\log N)^{-J(Z)/2}.
\end{equation}

\begin{introthm}[Theorem~\ref{thm:stack}, Corollary~\ref{cor:tail}; \status{proved}]\label{thmA}
For every fixed finite set $A$ of positive integers $a\equiv3\pmod4$,
\[
 \#\{p\le N:\ p\equiv1\ (\mathrm{mod}\ 24),\ -1\notin\Rat_a(x_a)\ \text{for all }a\in A\}
 \ll_A\frac N{(\log N)^{1+|A|/2}}.
\]
In particular $\#\{p\le N:\ p\equiv1\ (24),\ \amin(p)>Z\}\ll_Z N/(\log N)^{1+J(Z)/2}$
for every fixed $Z\ge3$.  A uniform version holds for $Z=o(\log\log N)$
(Theorem~\ref{thm:uniform}).
\end{introthm}

The key input is a \emph{half-set lemma} (Lemma~\ref{lem:halfset}): if a
window fails, all prime factors of $x_a$ lie in one of boundedly many
explicit sets of exactly half of the reduced classes modulo $a$.  Theorem~A
is then a fixed-dimension upper-bound sieve.

\begin{introthm}[Theorem~\ref{thm:W1}; \status{proved} modulo cited sieve theorems]\label{thmB}
\[
 \#\{p\le x:\ p\equiv1\ (\mathrm{mod}\ 840),\ \text{$(p+3)/4$ has no prime factor $\equiv2$ (mod 3)}\}
 \gg\frac{x}{(\log x)^{3/2}},
\]
and every prime counted is Mordell-hard with $\amin(p)\ge7$.
\end{introthm}

\begin{introthm}[Theorem~\ref{thm:W2}; \status{conditional} on Elliott--Halberstam]\label{thmC}
Assume the Elliott--Halberstam conjecture.  Then $\gg x/(\log x)^2$
Mordell-hard primes $p\le x$ have $\amin(p)\ge11$.
\end{introthm}

Together with Theorem~A these give the exact orders
\[
 \#\{p\le x:\ p\equiv1\ (24),\ \amin(p)\ge7\}\asymp\frac{x}{(\log x)^{3/2}},\qquad
 \#\{p\le x:\ p\equiv1\ (24),\ \amin(p)\ge11\}\asymp\frac{x}{(\log x)^{2}},
\]
the first modulo the cited sieve theorems, the second (lower bound only)
on Elliott--Halberstam (Corollary~\ref{cor:orders}).

The proofs of Theorems~B and~C rest on a congruence fact
(Lemma~\ref{lem:parity}): the number of prime factors of $x_a$ that are
non-residues modulo $a$, counted with multiplicity, has parity
$\leg pa$.  For one window and a sieve of dimension $1/2$ this removes the
parity obstruction.  The next result shows that this congruence input is
not optional.

\begin{introthm}[Theorem~\ref{thm:P1}; \status{proved}]\label{thmD}
Let $\mathcal A^{\pm}=\{p\le x:\ p\equiv1\ (8),\ p\equiv1\ (35),\ \leg p3=\pm1\}$
and $P_3=\{\ell\ \text{prime}:\ \ell\equiv2\ (3)\}$.  The two sequences
$\{(p+3)/4:\ p\in\mathcal A^{\pm}\}$ have the same sieve data with respect
to $P_3$ up to remainders admissible at level $x^{1/2}(\log x)^{-B}$
(Bombieri--Vinogradov), yet window~$3$ fails for no $p\in\mathcal A^-$
and for $\gg x/(\log x)^{3/2}$ primes $p\in\mathcal A^+$.
\end{introthm}

Two windows (3 and 7) form a sieve problem of dimension one, for which the
same mechanism needs a level of distribution close to~$1$; this is why
Theorem~C is conditional.  The problem is a window analogue of the
Friedlander--Iwaniec hyperbolic prime number theorem \cite{FI09}, whose
lower bound is likewise conditional (Lemma~\ref{lem:quadric},
Remark~\ref{rem:FI}).  Section~\ref{sec:model} records, with an explicit
\status{model-only} label, a finite linear-programming computation in a
coarse discretised model in which Type~I information of level $1/2$ plus
parity does not force both windows to fail.

\section{The window frame}\label{sec:frame}

\subsection{The window criterion}

\begin{lemma}[two unit fractions]\label{lem:two}
Let $r,s$ be coprime positive integers.  Then $r/s=1/y+1/z$ has a solution
in positive integers if and only if there are coprime positive integers
$d_1,d_2$ with $d_1d_2\mid s$ and $r\mid d_1+d_2$; the solution is then
$y=s(d_1+d_2)/(rd_1)$, $z=s(d_1+d_2)/(rd_2)$.
\end{lemma}

\begin{proof}
Sufficiency is the identity $1/y+1/z=r(d_1+d_2)/(s(d_1+d_2))$; both
denominators are integers because $d_1d_2\mid s$, $\gcd(d_1,d_2)=1$ and
$r\mid d_1+d_2$, $\gcd(r,d_i)\mid\gcd(r,s)=1$.  Conversely, write $y=gy_1$,
$z=gz_1$ with $\gcd(y_1,z_1)=1$.  Then $rgy_1z_1=s(y_1+z_1)$.  As
$\gcd(y_1+z_1,y_1z_1)=1$, we get $y_1z_1\mid s$; and as $\gcd(r,s)=1$, we
get $r\mid y_1+z_1$.  Take $d_1=z_1$, $d_2=y_1$.
\end{proof}

\begin{theorem}[window criterion; \status{proved}]\label{thm:frame}
Let $p\equiv1\pmod4$ be prime, let $a\equiv3\pmod4$ be positive, put
$x=x_a=(p+a)/4$ and suppose $p\nmid x$.  Then $\gcd(a,x)=1$, and:
\begin{enumerate}
\item If $u,v\ge1$ are coprime, $uv\mid x$ and $u\equiv-v\pmod a$, then
 \begin{equation}\label{eq:typeII}
 \frac4p=\frac1x+\frac{au}{px(u+v)}+\frac{av}{px(u+v)}
 \end{equation}
 is a solution of \eqref{eq:ES} in positive integers.
\item If $u,v\ge1$ are coprime, $uv\mid x$ and $u\equiv-pv\pmod a$, then
 \begin{equation}\label{eq:typeI}
 \frac4p=\frac1x+\frac{au}{px(u+pv)}+\frac{av}{x(u+pv)}
 \end{equation}
 is a solution of \eqref{eq:ES} in positive integers.
\item Conversely, if $4/p=1/x+1/y+1/z$ with $p\nmid x$, then $x=x_a$ for
 $a=4x-p$, which is a positive integer $\equiv3\pmod4$, and
 $\Rat_a(x)\cap\{-1,-p\}\neq\varnothing$.
\end{enumerate}
Consequently \eqref{eq:ES} is solvable for $n=p$ if and only if
$\amin(p)<\infty$.
\end{theorem}

\begin{proof}
Since $4x-a=p$, $\gcd(a,x)=\gcd(p,x)=1$.  Also $a$ is odd and coprime to
$p$, so $4/p-1/x=a/(px)$ is in lowest terms.  By Lemma~\ref{lem:two} with
$(r,s)=(a,px)$, $a/(px)=1/y+1/z$ is solvable if and only if there are coprime
$d_1,d_2$ with $d_1d_2\mid px$ and $a\mid d_1+d_2$.  Taking $(d_1,d_2)=(u,v)$
gives \eqref{eq:typeII}, and $(d_1,d_2)=(u,pv)$ gives \eqref{eq:typeI}; in
both cases the denominators $px(d_1+d_2)/(ad_i)$ are integers by
Lemma~\ref{lem:two}, and one checks that the two displayed fractions are
$1/y,1/z$.  This proves (1) and (2).

For (3): $1/x<4/p$ gives $a=4x-p>0$, and $a\equiv-p\equiv3\pmod4$.
Lemma~\ref{lem:two} gives coprime $d_1,d_2$ with $d_1d_2\mid px$ and
$a\mid d_1+d_2$.  Since $\gcd(d_1,d_2)=1$ and $p\,\|\,px$, the prime $p$
divides at most one of them.  If $p\nmid d_1d_2$, then $d_1d_2\mid x$ and
$d_1\equiv-d_2\pmod a$, so $-1\in\Rat_a(x)$.  If $p\mid d_2$, say
$d_2=pv$, then $d_1v\mid x$, $\gcd(d_1,v)=1$ and $d_1\equiv-pv$, so $d_1v^{-1}\equiv-p$ and
$-p\in\Rat_a(x)$; the case $p\mid d_1$ is the same with $d_1,d_2$
exchanged.

Finally, if \eqref{eq:ES} holds for $n=p$, not all of $x,y,z$ are
divisible by $p$: otherwise multiplying by $p$ would write $4$ as a sum of
three unit fractions.  Relabel so that $p\nmid x$ and apply (3); then
window $a=4x-p$ succeeds.  The converse is (1)--(2).
\end{proof}

Solutions from \eqref{eq:typeII} have two denominators divisible by $p$
and those from \eqref{eq:typeI} exactly one; these are the Type~II and
Type~I solutions of the literature \cite{ET}.  The criterion is a
reformulation of the classical parametrisations of these two types
(compare \cite[\S2]{ET}); we use it only in the form of
Theorem~\ref{thm:frame}.  We do not use the targets $-p$ in any upper
bound below: all upper bounds concern the larger event that $-1\notin
\Rat_a(x_a)$.

\subsection{Window reciprocity and the parity of the bad part}

\begin{lemma}[window reciprocity; \status{proved}]\label{lem:recip}
Let $p\equiv1\pmod8$ be prime, $a\equiv3\pmod4$ with $0<a<3p$, and
$x=x_a$.  Every prime $r\mid x$ is coprime to $a$ and satisfies
$\leg ra=\leg rp$.
\end{lemma}

\begin{proof}
If $r\mid a$ then $r\mid4x-a=p$, so $r=p$; but $r\le x<p$.  If $r$ is odd,
Jacobi reciprocity with $(a-1)/2$ odd gives
$\leg ra=\leg{-1}r\leg ar=\leg{-a}r$; since $r\mid p+a$ this is $\leg pr$,
which equals $\leg rp$ as $p\equiv1\pmod4$.  If $r=2$ then $8\mid p+a$, so
$a\equiv7\pmod8$ and $\leg2a=1=\leg2p$.
\end{proof}

Call a prime $r\nmid a$ \emph{$a$-bad} if $\leg ra=-1$ and \emph{$a$-good}
otherwise, and let $\Omega_a^-(n)$ be the number of $a$-bad prime factors
of $n$ counted with multiplicity.  The set of $a$-bad primes is a union of
reduced classes modulo $4a$, of relative density $1/2$, and does not
depend on~$p$.

\begin{lemma}[clean windows fail; \status{proved}]\label{lem:F1}
Let $p\equiv1\pmod 8$, $a\equiv3\pmod4$, $0<a<3p$.  If $x_a$ has no
$a$-bad prime factor, window $a$ fails.
\end{lemma}

\begin{proof}
$u\mapsto\leg ua$ is a character of $(\Z/a\Z)^\times$, so every element of
$\Rat_a(x_a)$ has symbol $+1$.  But $\leg{-1}a=-1$ as $a\equiv3\pmod4$, and
$\leg{-p}a=-\leg{4x_a}a=-\prod_{r\mid x_a}\leg ra^{v_r(x_a)}=-1$.
\end{proof}

By Lemma~\ref{lem:recip}, ``$x_a$ has no $a$-bad prime factor'' is the same as
``every prime factor of $x_a$ is a quadratic residue modulo $p$''.  We call
such a window \emph{clean}.  Writing the condition through $\leg\cdot a$
rather than $\leg\cdot p$ makes it a sifting condition on the shifted prime
$x_a$ by a fixed set of primes of density $1/2$.

\begin{lemma}[parity of the bad part; \status{proved}]\label{lem:parity}
If $\gcd(x_a,a)=1$ then $(-1)^{\Omega_a^-(x_a)}=\leg{x_a}a=\leg pa$.
\end{lemma}

\begin{proof}
Multiplicativity gives $\leg{x_a}a=\prod_r\leg ra^{v_r(x_a)}=(-1)^{\Omega_a^-(x_a)}$,
and $4x_a\equiv p\pmod a$ with $\leg4a=1$.
\end{proof}

\begin{lemma}[window 3]\label{lem:w3}
Let $p\equiv1\pmod{24}$.  Window $3$ fails if and only if $x_3=(p+3)/4$ has
no prime factor $\equiv2\pmod3$.  Hence $\amin(p)\ge7$ exactly in this case.
\end{lemma}

\begin{proof}
Modulo $3$ both targets equal $2$ ($-p\equiv-1$).  The prime $3$ does not
divide $x_3$ as $p\equiv1\pmod3$.  $\Rat_3(x_3)$ is generated by the classes
of the prime factors of $x_3$, so it contains $2$ if and only if some prime
factor is $\equiv2\pmod3$.  (The ``if'' direction is also Lemma~\ref{lem:F1}.)
\end{proof}

\section{The half-set lemma}\label{sec:halfset}

Let $a\equiv3\pmod4$, $a\ge3$, and $G=(\Z/a\Z)^\times$.  Since some prime
factor of $a$ is $\equiv3\pmod4$, $-1$ is not a square in $G$; and $g\ne-g$
as $a$ is odd.  The Klein four-group
$\{g\mapsto g,\ g\mapsto g^{-1},\ g\mapsto-g,\ g\mapsto-g^{-1}\}$ acts on~$G$,
and its orbits are of two kinds:
\begin{itemize}
\item $4$-orbits $\{g,g^{-1},-g,-g^{-1}\}$ when $g^2\ne1$ (the four
 elements are distinct because $g^2\neq\pm1$);
\item $2$-orbits $\{g,-g\}$ when $g^2=1$; one of them is $\{1,-1\}$.
\end{itemize}
Split every $4$-orbit as $\{g,g^{-1}\}\sqcup\{-g,-g^{-1}\}$ and every
$2$-orbit as $\{g\}\sqcup\{-g\}$.  A \emph{selection} $\sigma$ chooses one
half of every orbit, subject to choosing $\{1\}$ from $\{1,-1\}$; let
$S_\sigma\subseteq G$ be the union of the chosen halves.  Since $a$ is odd,
$g^2=1$ has exactly $2^{\omega(a)}$ solutions, so there are
$(\varphi(a)-2^{\omega(a)})/4$ orbits of size four and $2^{\omega(a)-1}$ of
size two.  Hence
\begin{equation}\label{eq:beta}
 |S_\sigma|=\tfrac12\varphi(a)\ \text{for every }\sigma,\qquad
 \#\{\sigma\}=2^{\beta(a)},\quad
 \beta(a)=\frac{\varphi(a)-2^{\omega(a)}}4+2^{\omega(a)-1}-1\le\frac{\varphi(a)}4+2^{\omega(a)-2}.
\end{equation}
For $x$ coprime to $a$ let $C(x)\subseteq G$ be the set of classes of the
prime factors of~$x$.

\begin{lemma}[half-set lemma; \status{proved}]\label{lem:halfset}
Let $a\equiv3\pmod4$, $a\ge3$, and $\gcd(x,a)=1$.  If $-1\notin\Rat_a(x)$,
then $C(x)\subseteq S_\sigma$ for some selection~$\sigma$.  In particular this
holds whenever window $a$ fails and $p\nmid x_a$, with $x=x_a$.
\end{lemma}

\begin{proof}
For distinct primes $r,s\mid x$ the set $\Rat_a(x)$ contains $r$, $rs$ and
$rs^{-1}$ (take $(u,v)=(r,1),(rs,1),(r,s)$).  Let $g,h\in C(x)$, possibly
equal.  If $g,h$ are the classes of distinct primes, then $-1\notin\Rat_a(x)$
gives $gh\ne-1$ and $gh^{-1}\ne-1$.  If $g=h$, then $gh=g^2\ne-1$ because
$-1$ is not a square, and $gh^{-1}=1\neq-1$.  Also $-1\notin C(x)$.  Hence
\[
 C(x)\cap(-C(x))=\varnothing,\qquad C(x)\cap(-C(x)^{-1})=\varnothing,\qquad -1\notin C(x).
\]
In a $4$-orbit, an element $g\in C(x)$ excludes $-g$ and $-g^{-1}$ from
$C(x)$, and so does $g^{-1}\in C(x)$; so $C(x)$ meets the orbit inside one
half.  In a $2$-orbit $\{g,-g\}$ the two elements exclude each other, and
in $\{1,-1\}$ the element $-1$ is excluded.  Choose $\sigma$ accordingly,
arbitrarily on orbits that $C(x)$ misses.
\end{proof}

\begin{remark}\label{rem:halfset}
(i) For prime $a$ the selection with $S_\sigma$ equal to the squares is the
clean case of Lemma~\ref{lem:F1}.  Failures need not be clean: for
$a=7$, $x=17$, $p=4\cdot17-7=61$ one has $C(x)=\{3\}$ and
$\Rat_7(17)=\{1,3,5\}$, which contains neither $-1\equiv6$ nor
$-p\equiv2$, although $\leg{17}7=-1$.  Such a failure is confined to the
non-subgroup half-set $\{1,3,5\}$.
(ii) The lemma has no exceptions coming from exponent multiplicities and
holds for composite $a$.  Every failure is a confinement of all prime
factors to a set of exactly half of the classes; this is what makes the
exponent in Theorem~\ref{thm:stack} exact.
(iii) Only the target $-1$ and products of at most two primes were used.
\end{remark}

\section{Upper bounds: fixed-set stacking}\label{sec:stack}

\subsection{A large-sieve lemma}

\begin{lemma}\label{lem:LS}
Let $I$ be an interval containing $X\ge2$ integers, $\mathcal P$ a finite set of
primes, and for $\ell\in\mathcal P$ let $\Omega_\ell$ be a set of $\nu(\ell)<\ell$
residue classes modulo~$\ell$.  Let $S\subseteq I$ avoid $\Omega_\ell$ modulo $\ell$
for every $\ell\in\mathcal P$.  Put
$V=\prod_{\ell\in\mathcal P}(1-\nu(\ell)/\ell)$ and
$\Lambda=\sum_{\ell\in\mathcal P}\nu(\ell)\log\ell/\ell$.  If
$\Lambda\le(\log X)/4$, then $|S|\le4XV$.
\end{lemma}

\begin{proof}
By the arithmetic large sieve (Cited Theorem~\ref{cit:LS}), for every
$Q\ge1$,
\[
 |S|\le\frac{X+Q^2}{H(Q)},\qquad H(Q)=\sum_{q\le Q}g(q),\qquad
 g(q)=\mu^2(q)\prod_{\ell\mid q}\frac{\nu(\ell)}{\ell-\nu(\ell)},
\]
the sum being over $q$ composed of primes of $\mathcal P$.  The full sum is
$\sum_q g(q)=\prod_{\ell\in\mathcal P}(1+\nu(\ell)/(\ell-\nu(\ell)))=V^{-1}$.
The weights $Vg(q)$ are the law of the random squarefree number
$q=\prod_{\ell\in\mathcal P,\ \xi_\ell=1}\ell$, where the $\xi_\ell$ are
independent Bernoulli variables with $\Pr(\xi_\ell=1)=\nu(\ell)/\ell$.  Then
$\mathbb E\log q=\Lambda$, and Markov's inequality gives
$\Pr(\log q>\log Q)\le\Lambda/\log Q\le\frac12$ for $Q=X^{1/2}$.  Hence
$H(X^{1/2})\ge V^{-1}/2$ and $|S|\le 2X\cdot2V$.
\end{proof}

\subsection{The stacking theorem}

\begin{theorem}[fixed-set stacking; \status{proved}]\label{thm:stack}
Let $A$ be a fixed finite set of positive integers $a\equiv3\pmod4$, and
$J=|A|$.  Then
\[
 \#\{p\le N:\ p\equiv1\ (\mathrm{mod}\ 24),\ -1\notin\Rat_a(x_a)\ \text{for every }a\in A\}
 \ll_A\frac{N}{(\log N)^{1+J/2}}.
\]
\end{theorem}

\begin{proof}
It suffices to count $p\in(N/2,N]$ with $N$ large and sum over dyadic
ranges.  For $p>3\max A$ we have $x_a<p$ and $\gcd(x_a,a)=\gcd(p,a)=1$, so
Lemma~\ref{lem:halfset} applies: each counted $p$ has a selection vector
$\sigma=(\sigma_a)_{a\in A}$ with $C(x_a)\subseteq S_{\sigma_a}$ for every
$a\in A$.  There are $\prod_{a\in A}2^{\beta(a)}=O_A(1)$ vectors; fix one.

Write $p=24t+1$; then $t$ runs over an interval $I$ with
$X\asymp N$ integers.  Put $y=2\max A+24$, $\log z=(\log X)/(5(J+1))$,
$\mathcal P=\{\ell\ \text{prime}:\ y<\ell\le z\}$, and let $\Omega_\ell$
consist of the class $24t+1\equiv0\pmod\ell$ and, for every $a\in A$ with
$(\ell\bmod a)\notin S_{\sigma_a}$, the class $24t+1+a\equiv0\pmod\ell$.
Every counted $t$ avoids $\Omega_\ell$: $\ell\le z<N/2<p$ gives $\ell\nmid p$;
and if $\ell\mid p+a$ then, $\ell$ being odd, $\ell\mid x_a$, so
$(\ell\bmod a)\in C(x_a)\subseteq S_{\sigma_a}$.  The listed classes are
distinct, since $\ell>y$ exceeds every $a$ and every $|a-a'|$, and
$\ell\nmid24$.  Thus $\nu(\ell)=1+\#\{a\in A:\ (\ell\bmod a)\notin
S_{\sigma_a}\}\le J+1<\ell$, and
$\Lambda\le(J+1)\sum_{\ell\le z}\log\ell/\ell\le(J+1)(\log z+O(1))\le(\log X)/4$
for large $N$.

Each $G\setminus S_{\sigma_a}$ is a union of $\varphi(a)/2$ reduced classes
modulo $a$, so Mertens' theorem in arithmetic progressions to the fixed
moduli $a\in A$ (Cited Theorem~\ref{cit:mertens}) gives
\[
 \sum_{\ell\in\mathcal P}\frac{\nu(\ell)}\ell
 =\Bigl(1+\frac J2\Bigr)\log\log z+O_A(1)
 =\Bigl(1+\frac J2\Bigr)\log\log N+O_A(1).
\]
By Lemma~\ref{lem:LS},
$|S|\le4XV\le4X\exp(-\sum_{\ell\in\mathcal P}\nu(\ell)/\ell)\ll_A N(\log N)^{-1-J/2}$.
Summing over the $O_A(1)$ vectors $\sigma$ and over dyadic ranges
(primes $p\le\sqrt N$ contribute trivially) proves the theorem.
\end{proof}

\begin{corollary}[tail of $\amin$; \status{proved}]\label{cor:tail}
For every fixed $Z\ge3$,
\[
 \#\{p\le N:\ p\equiv1\ (\mathrm{mod}\ 24),\ \amin(p)>Z\}\ll_Z\frac N{(\log N)^{1+J(Z)/2}}
 =\pi(N)(\log N)^{-J(Z)/2}\cdot O_Z(1).
\]
\end{corollary}

\begin{proof}
For $p>3Z$ every window $a\le Z$ has $p\nmid x_a$; $\amin(p)>Z$ means
that all of them fail, in particular $-1\notin\Rat_a(x_a)$.  Apply
Theorem~\ref{thm:stack} with $A=\{a\le Z:\ a\equiv3\ (4)\}$.
\end{proof}

For example, the number of $p\le N$, $p\equiv1\pmod{24}$, with
$\amin(p)>7$, $>11$, $>23$ is $\ll N/(\log N)^{2}$,
$\ll N/(\log N)^{5/2}$, $\ll N/(\log N)^{4}$ respectively.  Running the
same argument with $p=4t+1$ covers all $p\equiv1\pmod4$; we restrict to
$p\equiv1\pmod{24}$ because the other primes are settled by Mordell's
identities.

\subsection{Uniformity in $Z$}

The union over selection vectors costs $2^{\beta_{\rm tot}(Z)}$ with
$\beta_{\rm tot}(Z)=\sum_{a\le Z,\,a\equiv3\,(4)}\beta(a)=O(Z^2)$, which
is affordable while $Z\ll\log\log N$.

\begin{theorem}[uniform stacking; \status{proved} modulo Siegel--Walfisz]\label{thm:uniform}
Let $L=\log\log N$, $\mathcal L=\log N$, and fix $C_0\ge1$.  Uniformly for
$3\le Z\le C_0L$ and $N\ge N_0(C_0)$, with $J=J(Z)$,
\[
 \#\{N/2<p\le N:\ p\equiv1\ (24),\ \amin(p)>Z\}
 \le\frac N{\mathcal L}\exp\Bigl(-\frac J2\bigl(L-3\log L-C\bigr)+\beta_{\rm tot}(Z)\log2+O(\log L)\Bigr),
\]
where $C$ and the implied constant depend only on $C_0$.  In particular the
left side is $\pi(N)(\log N)^{-(1/2-o(1))J(Z)}$ uniformly for $Z=o(L)$.
\end{theorem}

\begin{proof}
Repeat the proof of Theorem~\ref{thm:stack} with $y=\exp(L^2)$ and
$\log z=\mathcal L/(5(J+1))$.  Now $y>2Z+24$, so the classes are distinct
and $\nu(\ell)\le J+1<\ell$; and $\Lambda\le(J+1)(\log z+O(1))\le\mathcal L/4$
because $J\le C_0L$.  The moduli satisfy $a\le C_0L=C_0(\log y)^{1/2}$, so
the Siegel--Walfisz theorem (Cited Theorem~\ref{cit:SW}) and partial summation give
\[
 \sum_{\substack{y<\ell\le z\\ \ell\equiv c\ (a)}}\frac1\ell
 =\frac1{\varphi(a)}\log\frac{\log z}{\log y}+O\bigl(e^{-c_1L}\bigr)
\]
uniformly in $a\le Z$ and $\gcd(c,a)=1$, with $c_1=c_1(C_0)>0$.  Summing over
at most $\sum_{a\le Z}\varphi(a)/2=O(L^2)$ classes, the total error is
$o(1)$, and
$\sum_{\ell\in\mathcal P}\nu(\ell)/\ell=(1+J/2)\log(\log z/\log y)+o(1)$.
Here $\log(\log z/\log y)=L-2\log L-\log(5(J+1))\ge L-3\log L-C$.  Lemma~\ref{lem:LS}
bounds the count for one selection vector by
$4X\exp(-(1+J/2)(L-3\log L-C)+o(1))$, and $e^{-(L-3\log L-C)}=e^{C}L^3/\mathcal L$.
Summing over the $2^{\beta_{\rm tot}(Z)}$ selection vectors gives the
bound.  For the last claim note $\beta_{\rm tot}(Z)=O(Z^2)=o(JL)$ when $Z=o(L)$.
\end{proof}

\begin{remark}
Since $\sum_{a\le Z,\,a\equiv3\,(4)}\varphi(a)=(1+o(1))Z^2/\pi^2$, one has
$\beta_{\rm tot}(Z)=(1+o(1))Z^2/(4\pi^2)$, and optimising the exponent
$-(Z/8)L+(\log2/4\pi^2)Z^2$ at $Z\approx\pi^2L/(4\log2)\approx3.56L$ (inside the
range with $C_0=4$) gives
$\#\{p\le N:\ p\equiv1\ (24),\ \amin(p)>Z\}\le N\exp\bigl(-(\pi^2/(64\log2)-o(1))(\log\log N)^2\bigr)$.
For windows up to $(\log N)^\theta$ with $\theta<\log3/(1+\log3)$, stronger
tails of the shape $N\exp(-c(\log N)^\theta\log\log N)$ follow from an
earlier unpublished argument in this project (\cite[\S14]{Notes}); we do
not reproduce it.  None of these bounds approaches $\amin(p)<\infty$ for all $p$.
\end{remark}

\section{Cited results}\label{sec:cited}

We list every external result used, with a note on what was checked.
``Checked'' means that the exact statement was compared with an archived
copy of the source; ``from memory'' means it was not, and a referee should
compare it with the reference given.

For a finite sequence of nonnegative weights $(a_n)$, a set of primes
$\mathcal P$ and $z\ge2$, put $P(z)=\prod_{\ell\in\mathcal P,\,\ell<z}\ell$,
$S(\mathcal A,\mathcal P,z)=\sum_{(n,P(z))=1}a_n$ and
$A_d=\sum_{d\mid n}a_n$.  Suppose that for squarefree $d\mid P(z)$
\begin{equation}\label{eq:sievedata}
 A_d=g(d)X+r_d,
\end{equation}
with $g$ multiplicative, $0\le g(\ell)<1$, and put
$V(z)=\prod_{\ell\in\mathcal P,\,\ell<z}(1-g(\ell))$.  The \emph{dimension
condition} $(\Omega_\kappa)$ is
\[
 \prod_{\substack{w\le\ell<z\\ \ell\in\mathcal P}}(1-g(\ell))^{-1}
 \le\Bigl(\frac{\log z}{\log w}\Bigr)^{\kappa}\Bigl(1+\frac K{\log w}\Bigr)
 \qquad(2\le w<z).
\]

\begin{citedthm}[arithmetic large sieve]\label{cit:LS}
Let $S$ be a set of integers in an interval of length $X$ which, for every
prime $\ell$ in a finite set $\mathcal P$, avoids $\nu(\ell)<\ell$ residue
classes modulo $\ell$.  Then for $Q\ge1$, $|S|\le(X+Q^2)/H(Q)$ with $H(Q)$
as in the proof of Lemma~\ref{lem:LS}.
\end{citedthm}
\noindent Montgomery \cite{Montgomery68}, with the constant $X+Q^2$ from the
Montgomery--Vaughan form of the analytic large sieve \cite{MVLS}; see also
\cite[\S9.4]{OdC}.  \emph{From memory}; it is the standard form and is
used in the same way in \cite[Lemma 12.1]{Notes}.

\begin{citedthm}[Mertens in progressions]\label{cit:mertens}
For fixed $a\ge1$ and $\gcd(c,a)=1$,
$\sum_{\ell\le z,\ \ell\equiv c\,(a)}1/\ell=\varphi(a)^{-1}\log\log z+M(a,c)+O_a(1/\log z)$.
Consequently, if $\mathcal P$ is a union of $k$ reduced classes modulo $a$,
then $\prod_{\ell<w,\ \ell\in\mathcal P}(1-1/\ell)^{-1}=c_{\mathcal P}(\log w)^{k/\varphi(a)}(1+O(1/\log w))$
with $c_{\mathcal P}>0$.
\end{citedthm}
\noindent Standard consequence of the prime number theorem in
arithmetic progressions; e.g.\ \cite[\S4.2]{MV}.

\begin{citedthm}[Siegel--Walfisz]\label{cit:SW}
For every $B>0$ there is $c_B>0$ such that
$\pi(t;a,c)=\li(t)/\varphi(a)+O_B(t\exp(-c_B\sqrt{\log t}))$ uniformly for
$a\le(\log t)^B$ and $\gcd(c,a)=1$.
\end{citedthm}
\noindent \cite[Cor.~11.21]{MV}.  Standard; \emph{from memory} as to the
numbering.  The constants are ineffective.

\begin{citedthm}[Bombieri--Vinogradov]\label{cit:BV}
For every $A>0$ there is $B=B(A)$ such that
$\sum_{k\le x^{1/2}(\log x)^{-B}}\max_{\gcd(b,k)=1}|\pi(x;k,b)-\li(x)/\varphi(k)|\ll_A x(\log x)^{-A}$.
\end{citedthm}
\noindent Standard (e.g.\ \cite[Thm.~17.1]{IK}); the version with
$\max_{y\le x}$ is not needed.

\begin{definition}[Elliott--Halberstam conjecture, EH]\label{def:EH}
For every $\eta,A>0$:
$\sum_{k\le x^{1-\eta}}\max_{\gcd(b,k)=1}|\pi(x;k,b)-\li(x)/\varphi(k)|\ll_{\eta,A}x(\log x)^{-A}$.
\end{definition}

\begin{citedthm}[semi-linear sieve, lower bound]\label{cit:semi}
Assume \eqref{eq:sievedata} and $(\Omega_{1/2})$.  Let $D=z^s$ with
$1\le s\le2$.  Then
\[
 S(\mathcal A,\mathcal P,z)\ge XV(z)\bigl(f(s)+o(1)\bigr)-\sum_{d<D,\ d\mid P(z)}|r_d|,
 \qquad f(s)=\Bigl(\frac{e^\gamma}{\pi s}\Bigr)^{1/2}\int_1^s\frac{dt}{\sqrt{t(t-1)}},
\]
where $o(1)\to0$ as $D\to\infty$, uniformly for $s\in[1,2]$, depending
only on~$K$.
\end{citedthm}
\noindent Iwaniec \cite{Iw76}; in the $\beta$-sieve form
\cite[Thm.~11.13]{OdC} with $\kappa=1/2$, $\beta=1$.  \emph{Partly checked:}
the archived paper of Ter\"av\"ainen \cite[(6.4)]{Ter} applies
\cite[Thm.~11.13]{OdC} with exactly this $f$ and $s\in[1,2]$ (with error
$O((\log N)^{-0.1})$ and the remainder weighted by the lower-bound sieve
weights, which are bounded by $1$).  The general hypotheses as stated
above are the standard ones and were \emph{not} compared with
\cite{OdC} or \cite{Iw76}.  We use only that, for $1<s\le2$,
\begin{equation}\label{eq:fsemi}
 f(s)\ge c_f(s-1)^{1/2},\qquad c_f=(e^\gamma/\pi)^{1/2},
\end{equation}
which follows from $t(t-1)\le2(t-1)$ on $[1,2]$:
$\int_1^s dt/\sqrt{t(t-1)}\ge\int_1^s dt/\sqrt{2(t-1)}=\sqrt2\,(s-1)^{1/2}$, and
$(e^\gamma/(\pi s))^{1/2}\ge(e^\gamma/(2\pi))^{1/2}$.

\begin{citedthm}[linear sieve, lower bound]\label{cit:lin}
Assume \eqref{eq:sievedata} and $(\Omega_1)$.  Let $D=z^s$ with $2\le s\le3$.
Then
\[
 S(\mathcal A,\mathcal P,z)\ge XV(z)\bigl(f_1(s)+o(1)\bigr)-\sum_{d<D,\ d\mid P(z)}|r_d|,
 \qquad f_1(s)=\frac{2e^\gamma\log(s-1)}{s},
\]
with $o(1)\to0$ as $D\to\infty$, uniformly in $s\in[2,3]$, depending only on~$K$.
\end{citedthm}
\noindent Jurkat--Richert; \cite[Thm.~8.4]{HR}, \cite[Thm.~11.13]{OdC}
with $\kappa=1$, $\beta=2$.  \emph{From memory} (the archived source
\cite[(6.6)]{Ter} quotes only the upper linear bound).  We use only
$\log(1+u)\ge u/2$ on $[0,1]$, whence
\begin{equation}\label{eq:flin}
 f_1(s)\ge\tfrac{e^\gamma}3(s-2)\qquad(2<s\le3).
\end{equation}

\begin{citedthm}[Fuchs--Hsu--Rickards--Schindler--Stange]\label{cit:FHRSS}
Let $f=ax^2+bxy+cy^2$ be a primitive positive definite integral form of
discriminant $D$ with $\gcd(a,2D)=1$; let $A\ne0$, $B\ge1$ be integers with
$\gcd(A,B)=1$, and assume $2\mid AB$ or $D\not\equiv5\pmod8$.  Let $m\ge1$,
$\gcd(l,m)=1$, $\gcd(m,2DB)=1$, $\gcd(l-A,m)=1$.  Then the number of primes
$p\le N$, $p\equiv l\pmod m$, primitively represented by $Bf(x,y)+A$ is
$\gg N/(\log N)^{3/2}$.
\end{citedthm}
\noindent \cite[Thm.~1.1(2)]{FHRSS}.  \emph{Checked} against the
archived preprint (statement only; the proof was not checked).  Used only
in Remark~\ref{rem:FHRSS} as a cross-check.

The proofs below use the sieve theorems only through
\eqref{eq:fsemi} and \eqref{eq:flin}; all upper-bound sieving is done with
Lemma~\ref{lem:LS}, which is proved above from Cited
Theorem~\ref{cit:LS}.

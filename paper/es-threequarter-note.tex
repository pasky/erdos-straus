\title[A three-quarter exceptional-set bound for Erd\H{o}s--Straus]{A three-quarter exceptional-set bound\\ for the Erd\H{o}s--Straus equation}
\author{Anonymous}
\date{Internal research note, version C2. Internally proved; not externally refereed.}
\begin{document}
\begin{abstract}
Let $E(N)$ be the number of $n\leq N$ for which $4/n$ is not a sum of three
unit fractions.  Vaughan (1970) proved $E(N)\ll N\exp\{-c(\log N)^{2/3}\}$.
We prove
\[
 E(N)\ll N\exp\{-c(\log N)^{3/4}\},
\]
with an absolute, ineffective constant $c>0$.  We are not aware of an
earlier improvement of the exponent $2/3$.  The forced residue classes are
those of the classical Type~II identity, but a prime $\ell\leq X$ is used
together with a \emph{multiplier} $k\leq X^\kappa$, giving classes modulo
$k\ell$.  Conditionally on $n\bmod\lcm(k)$ the hits at different primes are
independent, and Bombieri--Vinogradov, Shiu's theorem and a Cauchy--Schwarz
argument show that every conditional fibre has expected number of hits
$\asymp(\log X)^3$, against $(\log X)^2$ for Vaughan's classes.  Instead of
the large sieve we count on $[1,N]$ an explicit nonnegative majorant made
of congruence classes, namely a small-prime selector times an even
Bonferroni truncation.  Its coefficient ledger costs
$\exp\{O((\log X)^4)\}$, and balancing gives the exponent $3/4$.  This note
has been checked internally only; it is not a proof of the Erd\H{o}s--Straus
conjecture.
\end{abstract}

\section{Introduction}\label{sec:intro}
Call a positive integer $n$ \emph{representable} if there are positive
integers $x_1,x_2,x_3$, not necessarily distinct, with
\begin{equation}\label{eq:ES}
 \frac4n=\frac1{x_1}+\frac1{x_2}+\frac1{x_3}.
\end{equation}
The Erd\H{o}s--Straus conjecture asserts that every $n\geq2$ is
representable.  Put
\[
 E(N)=\#\{1\leq n\leq N:n\text{ not representable}\},\qquad
 E_{\rm pr}(N)=\#\{p\leq N\text{ prime}:p\text{ not representable}\}.
\]
(Here $n=1$ is not representable; this is irrelevant for a density bound.)

\begin{theorem}\label{thm:main}
There are absolute constants $c_*,C_*>0$ such that for every real $N\geq3$
\[
 E_{\rm pr}(N)\leq E(N)\leq C_*N\exp\{-c_*(\log N)^{3/4}\}.
\]
The constants are ineffective, because the proof uses the
Bombieri--Vinogradov theorem.
\end{theorem}

\subsection*{Background}
Vaughan \cite{Vaughan} proved $E(N)\ll N\exp\{-c(\log N)^{2/3}\}$.  This
is still the bound quoted by Elsholtz and Tao \cite[\S1]{ET} and in the
survey of Bloom and Elsholtz \cite{BE}.  Pomerance and Weingartner
\cite[Thm.~1.3]{PW} give a version that is uniform in the numerator $m$:
for $4\leq m\leq(\log N)^2$, at most $N\exp\{-C(\log^2N/\varphi(m))^{1/3}\}$
integers $n\leq N$ have $m/n$ not a sum of three unit fractions.  For fixed
$m$ the exponent is again $2/3$.  For sums of $k\geq3$ unit fractions,
Elsholtz's bound has exponent $1-1/(2^{k-1}-1)$ \cite{Elsholtz},
\cite[\S on exceptional sets]{BE}.  This is $2/3$ for $k=3$; larger $k$ is
a different problem.  A companion internal note \cite{LL} obtains the
intermediate bound $N\exp\{-c(\log N)^{2/3}(\log\log N)^{1/3}\}$ by the
large sieve with multipliers $k\ll\log N$.  A search of the literature up
to 2026 found no improvement of the exponent $2/3$ for three unit fractions.
This is a report of a search, not a priority claim.

\subsection*{Where the exponents come from}
For a prime $\ell\equiv3\pmod4$, every divisor pair of $(\ell+1)/4$
forces a residue class modulo $\ell$ to consist of representable integers.
On average over primes $\ell\leq X$ the mass of these classes is
$\sum_\ell\omega(\ell)/\ell\asymp(\log X)^2$.  To turn mass $\mu$ at
primes $\leq X$ into a saving $\e^{-c\mu}$, any sieve of this kind must
control intersections of about $\mu$ classes.  Such intersections have
moduli of size up to $X^{O(\mu)}$, and this costs $\exp\{O(\mu\log X)\}$
in the error terms.  With $t=\log X$ and $\mu\asymp t^{\mathcal B}$, the
balance $t^{\mathcal B+1}\asymp\log N$ gives a saving
$(\log N)^{\mathcal B/(\mathcal B+1)}$.  Vaughan's large sieve has
$\mathcal B=2$, hence the exponent $2/3$.

The point of this note is $\mathcal B=3$.  For a multiplier $k\equiv1\pmod4$
up to $K=X^\kappa$, factorizations of $(k\ell+1)/4$ force classes modulo
$k\ell$ (Lemma~\ref{lem:identity}).  Conditionally on $n\bmod k$, such a
class is a class modulo $\ell$.  Summed over $k$, the extra mass is the
factor $\sum_k\varphi(k)/k^2\asymp\log K\asymp t$
(Theorem~\ref{thm:supply}).  There is a difficulty: $L_K=\lcm\{k\leq K\}$
satisfies $\log L_K\asymp K$, so $L_K$ is far larger than $N$, so one cannot split $[1,N]$
into progressions modulo $L_K$ as \cite{LL} does.  We avoid any such
splitting.  The majorant of Section~\ref{sec:assembly} is a fixed finite
signed combination of congruence classes.  Its mean is computed exactly in
a finite CRT probability space.  The fibres modulo $L_K$ appear only
inside that computation, and its count on $[1,N]$ costs one unit per term.
The terms number $\exp\{O(t^4)\}$, so $t\asymp(\log N)^{1/4}$ and the
saving is $t^3\asymp(\log N)^{3/4}$.

\begin{remark}[the method limit]\label{rem:limit}
Internal, unrefereed companion work \cite{KA3,SL} shows that $3/4$ is
optimal for the architecture used here.  More precisely, consider any
nonnegative CRT majorant that is $\geq1$ on all integers avoiding a finite
family of forced classes (the classes
$\mathcal R(M)=\{-4D\bmod M:D\mid((M+1)/4)^2\}$ of
Remark~\ref{rem:supply}, and the $(a,D)$ and Case~A classes in the
notation of \cite{KA3}) and of selector classes $0\bmod p$, and that is used through
$N\E\nu+\sum|{\rm coefficients}|$ with coefficient sum $<N$.  Every such
majorant saves at most $C(\log N)^{3/4}$; see \cite[Cor.~4.2]{KA3} and
\cite[Thm.~10.14]{SL}.  The majorant of Theorem~\ref{thm:assembly} is of
this form.  These results do not cover large sieves of super-polynomial
level, hybrid interval methods, or non-CRT inputs.  Nothing in this remark
is used below.
\end{remark}

The proof below has been checked by internal audits only.  It has not been

argument proved the supply estimate of Section~\ref{sec:supply} by an

Both are superseded by the shorter arguments here and have been removed.

\subsection*{Parameters}
Fix $\kappa=1/480$; any fixed $0<\kappa<1/240$ would do.  Throughout,
\begin{equation}\label{eq:parameters}
 \begin{gathered}
 t=\log X,\quad K=\lfloor X^\kappa\rfloor,\quad H=K^{10},\\
 \K(K)=\{1\leq k\leq K:k\equiv1\pmod4\},\quad
 L_K=\lcm_{k\in\K(K)}k.
 \end{gathered}
\end{equation}
The constants are chosen in the following order.  First come the absolute
analytic constants (Shiu, Brun--Titchmarsh, Bombieri--Vinogradov with a
fixed exponent) and the lattice threshold $K_0$.  Then the void constant
$c_v$ is chosen (Section~\ref{sec:void}), then the selector constant $B$
in $y=Bt^3$, then the Bonferroni constant $D_B$, then the ledger constant
$C_0$, and finally $\alpha$ in $t=\alpha(\log N)^{1/4}$.  Every later
``sufficiently large'' threshold depends only on the constants already
chosen, and none depends on a residue class, a subfamily of multipliers,
a moment order, or $N$.  We write $A\ll B$ with constants that are
absolute in this sense.

The guiding distinction is between using prime distribution at scale $X$
to \emph{construct} classes and counting those classes on \emph{all}
integers up to $N$.  No prime-counting estimate is ever used at scale $N$.
Integer counts in residue classes, each with error at most one, take its
place.

\section{Forced classes and a fixed atom family}\label{sec:atoms}
\begin{lemma}[multiplier identity]\label{lem:identity}
Let $k,\ell$ be positive integers with $k\ell\equiv3\pmod4$, and let
$(k\ell+1)/4=uvw$ with positive integers $u,v,w$.  If $n\geq1$ satisfies
$nv\equiv-u\pmod{k\ell}$, then $s=(nv+u)/(k\ell)$ is a positive integer and
\[
 \frac4n=\frac1{suw}+\frac1{nsvw}+\frac1{nuvw}.
\]
Thus every positive integer in the class $-uv^{-1}\pmod{k\ell}$ is
representable.  The inverse exists because $uvw$ is coprime to $k\ell$.
\end{lemma}
\begin{proof}
Over the common denominator $nsuvw$ the numerators add up to
$nv+u+s=s(k\ell+1)=4suvw$.
\end{proof}
For $k=1$ these are the classes used by Vaughan; the solvability criterion
is classical in the form ``$4abd-1\mid b+nc$'' (see \cite[Prop.~2.7]{ET},
where it is attributed to \cite{Vaughan}).  The multiplier $k$ only changes
which moduli $M=k\ell$ are used.

\subsection*{Blocks and atoms}
Put $x_j=2^jX^{1/2}$ and use the disjoint blocks
\[
 I_j=(x_j,\min(2x_j,X)],\qquad
 0\leq j<\Bigl\lceil\frac{\log X}{2\log2}\Bigr\rceil,
\]
with $z_j=x_j^{1/6}$.  A block is \emph{full} if $2x_j\leq X$.  Lower
estimates use only full blocks, and upper estimates enlarge the last block
to a full one.  There are $\asymp t$ blocks, and $t/2\leq\log x_j\leq t$.
Since $\kappa<1/240$, for all sufficiently large $X$ every block satisfies
\begin{equation}\label{eq:floor}
 z_j\geq X^{1/12}>K^{20}=H^2,
 \qquad \Lambda_j:=\log(z_j/H)\geq\tfrac12\log z_j.
\end{equation}

Let $\A_X$ be the set of quadruples $A=(k,\ell,u,v)$ with
\begin{equation}\label{eq:atomdef}
 k\in\K(K),\quad \ell\in I_j\text{ prime for some }j,\quad H<u,v\leq z_j,
 \quad (u,v)=(uv,k)=1,\quad 4uv\mid k\ell+1.
\end{equation}
Divisibility forces $\ell\equiv3\pmod4$.  To each atom $A$ attach the class
\begin{equation}\label{eq:cylinder}
 E_A=\{n:n\equiv-uv^{-1}\pmod{k\ell}\},\qquad
 H_X(n)=\sum_{A\in\A_X}\ind_{E_A}(n).
\end{equation}
By Lemma~\ref{lem:identity} with $w=(k\ell+1)/(4uv)$, a non-representable
$n$ has $H_X(n)=0$.  The family $\A_X$ is fixed: it does not depend on $N$
or on any residue class.

\subsection*{The probability space}
Let $2\leq y<K$ and $P_y=\prod_{p\leq y}p$.  We use the uniform measure on
$\mathbb Z/\mathcal M\mathbb Z$, where
\begin{equation}\label{eq:space}
 \mathcal M=\lcm(L_K,P_y)\prod_{\substack{X^{1/2}<\ell\leq X\\\ell\
 \mathrm{prime}}}\ell .
\end{equation}
Every class $a\bmod q$ with $q\mid\mathcal M$ has probability $1/q$.  The
size of $\mathcal M$ is irrelevant: it only defines probabilities, and
$[1,N]$ is never partitioned modulo $\mathcal M$.  We write
$c=n\bmod L_K$ and call $\{n:n\equiv c\ (L_K)\}$ the \emph{fibre} of $c$.

\begin{lemma}[distinctness and conditional independence]\label{lem:CRT}
For all sufficiently large $X$ the following hold.
\begin{enumerate}
\item Distinct atoms with the same prime $\ell$ have distinct classes
modulo $\ell$.
\item Conditionally on $c=n\bmod L_K$, and on any event determined by
$n\bmod P_y$, the residues $n\bmod\ell$ for the primes
$X^{1/2}<\ell\leq X$ are independent and uniform.
\item An atom $A$ meets the fibre of $c$ if and only if $k\mid u+cv$.
In that case $E_A\cap\{n\equiv c\ (L_K)\}=\{n\equiv c\ (L_K),\
n\equiv-uv^{-1}\ (\ell)\}$.
\end{enumerate}
Consequently, in the fibre of $c$, $H_X=\sum_\ell I_\ell$ is a sum of
independent Bernoulli variables, with $\Prob(I_\ell=1\mid c)=f_c(\ell)/\ell$, where
\begin{equation}\label{eq:fc}
 f_c(\ell)=\#\{A\in\A_X:\ell_A=\ell,\ k_A\mid u_A+cv_A\}.
\end{equation}
\end{lemma}
\begin{proof}
(1) Let $(k,\ell,u,v)$ and $(k',\ell,u',v')$ have the same class modulo
$\ell$, with $\ell\in I_j$.  Then $\ell\mid uv'-u'v$ and
$|uv'-u'v|<z_j^2<x_j<\ell$, so $uv'=u'v$.  Both fractions are reduced, so
$(u,v)=(u',v')$.  Next $k\ell\equiv k'\ell\equiv-1\pmod{4uv}$ and
$(\ell,4uv)=1$ give $k\equiv k'\pmod{4uv}$.  Since $4uv>4H^2>K$, $k=k'$.
(2) The primes $\ell>X^{1/2}>K>y$ are coprime to $\lcm(L_K,P_y)$, so this
is the Chinese remainder theorem.  (3) Since $k\mid L_K$, the condition
$n\equiv-uv^{-1}\pmod k$ is determined by $c$, and it holds if and only if
$k\mid u+cv$.  By (1) the active atoms at $\ell$ occupy $f_c(\ell)$
distinct classes modulo $\ell$, and at most one of them contains $n$.
\end{proof}
Note that $H_X(n)=\sum_\ell I_\ell(n)$ holds identically and not just in
distribution: at each $\ell$ at most one atom can contain $n$.

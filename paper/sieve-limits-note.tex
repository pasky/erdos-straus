\title[Sieve limits for the Erd\H{o}s--Straus exceptional set]{Sieve limits for the Erd\H{o}s--Straus exceptional set:\\ why $3/4$ is sharp for congruence sieves}
\author{Anonymous}
\date{Research note; internally reviewed results only, not externally refereed}
\begin{document}
\begin{abstract}
Let $E(N)$ count the $n\leq N$ for which $4/n$ is not a sum of three
positive unit fractions.  The best internally proved bound in this
programme is $E(N)\ll N\exp\{-c(\log N)^{3/4}\}$.  We show that the
exponent $3/4$ is a hard ceiling for a precisely defined class of
congruence sieves.  The class consists of nonnegative CRT majorants that
are at least one on the whole avoider set of a family of forced
congruence classes, whose final bound has the form
$N\cdot\E\nu+\sum_i|a_i|$, and whose family moduli are at most
$N^{O(1)}$.  For families in which every modulus has a dominant prime
factor, and, for the classes $\cR(M)$, for families of gapped moduli, every
such sieve saves at most $O((\log N)^{3/4})$.  No power of $\log\log N$
can be gained.  The majorant of the $3/4$ note lies in this class, and the
large-sieve argument of the $2/3$-loglog note is capped in the same way at
its own exponent.  We list exactly what the theorems exclude.  For
Selberg-type majorants we prove a sieve limit with no structural
hypothesis, in terms of a noise-stability quantity, and a two-prime cap
with twin moduli that is conditional on an equidistribution hypothesis.
Every statement carries its status label.  All proofs were checked
internally only; nothing here has been externally refereed.  This is not a
proof of the Erd\H{o}s--Straus conjecture.
\end{abstract}

\section{Introduction and status}\label{sec:intro}

Call $n\geq1$ \emph{representable} if $4/n=1/x_1+1/x_2+1/x_3$ with
positive integers $x_i$, and let $E(N)$ count the non-representable
$n\leq N$.  The Erd\H{o}s--Straus conjecture asserts that every $n\geq2$ is
representable.  Vaughan~\cite{Vaughan} proved
$E(N)\ll N\exp\{-c(\log N)^{2/3}\}$ with Montgomery's large
sieve~\cite{Montgomery}.  Two companion notes of this programme
improve this.  The first~\cite{LL} gains a factor
$(\log\log N)^{1/3}$ in the exponent.  The second~\cite{TQ} proves
\[
 E(N)\ll N\exp\{-c(\log N)^{3/4}\},
\]
with label \emph{internally proved; not externally refereed}.  The $3/4$
note itself explains $3/4$ heuristically: the forced classes have
cubic logarithmic mass, and a mass-driven optimisation gives
$\theta=B/(B+1)$ with $B=3$ \cite[\S\ref*{sec:intro}]{TQ}.

\section{Forced classes}\label{sec:forced}

A residue class $b\pmod G$ is \emph{forced} if every positive integer
$n\equiv b\pmod G$ is representable.  We use three families of forced
classes.  For an integer $D\geq1$ put
$g(D)=\prod_pp^{\lceil v_p(D)/2\rceil}$, so that $g(D)\mid D\mid g(D)^2$
and, for every integer $x$, $D\mid x^2\iff g(D)\mid x$.

\begin{lemma}[multiplier identity; {\cite[Lemma~2.1]{LL}}; \lab{proved}]\label{lem:identity}
Let $M\equiv3\pmod4$, $A=(M+1)/4$, and $A=uvw$ with positive integers
$u,v,w$.  If $n\geq1$ and $nv\equiv-u\pmod M$, then $s=(nv+u)/M$ is a
positive integer and $4/n=1/(suw)+1/(nsvw)+1/(nuvw)$.
\end{lemma}
\begin{proof}
On the common denominator $nsuvw$ the numerators sum to
$nv+u+s=s(M+1)=4suvw$.
\end{proof}

\begin{lemma}[the classes $\cR(M)$; {\cite[Lemma~18.1]{Notes}}; \lab{proved}]\label{lem:RM}
For $M\equiv3\pmod4$ let $\cR(M)$ be the set of classes $-uv^{-1}\pmod M$
over all factorisations $A=uvw$.  Then
\[
 \cR(M)=\{-4D\pmod M:\ D\mid A^2\},\qquad
 |\cR(M)|\leq\tau(A^2),
\]
and every class of $\cR(M)$ is forced.
\end{lemma}
\begin{proof}
Since $4A\equiv1\pmod M$, a factorisation $A=uvw$ gives
$-uv^{-1}\equiv-u^2wA^{-1}\equiv-4u^2w\pmod M$ with $u^2w\mid A^2$.
Conversely, if $A=\prod p^{e_p}$ and $D=\prod p^{d_p}$ with $d_p\leq2e_p$, put
$v_p(u)=\lfloor d_p/2\rfloor$, $v_p(w)=d_p-2\lfloor d_p/2\rfloor$ and
$v_p(v)=e_p-v_p(u)-v_p(w)\geq0$; then $uvw=A$ and $u^2w=D$.  Forcedness is
Lemma~\ref{lem:identity}.
\end{proof}

\begin{lemma}[$(a,D)$-classes; {\cite[Lemma~3.2]{ET}}; \lab{proved}]\label{lem:aD}
For all integers $a,D\geq1$ the class $n\equiv-(4D+a)\pmod{4a\,g(D)}$ is
forced.
\end{lemma}
\begin{proof}
Write $n+4D+a=4a\,g(D)\,j$ with $j\geq1$.  Then $M:=(n+4D)/a=4g(D)j-1$ is
$\equiv3\pmod4$ and $(M+1)/4=g(D)j$, so $D\mid g(D)^2\mid((M+1)/4)^2$.
Since $n\equiv-4D\pmod M$, $n$ lies in a class of $\cR(M)$, and
Lemma~\ref{lem:RM} applies.
\end{proof}
These classes encode exactly the Case-B witnesses of the classical
parametrisation \cite[Thm~3.1(B)]{Notes} (see \cite[Lemma~3.2]{ET}).

\begin{lemma}[Case-A classes; {\cite[\S3]{ET}}; \lab{proved}]\label{lem:caseA}
For $d\geq1$ and $m\mid4d+1$ (so $m$ is odd and $(m,4g(d))=1$), the class
$n\equiv-m^{-1}\pmod{4g(d)}$ is forced.
\end{lemma}
\begin{proof}
Put $z_0=(nm+1)/4$, an integer.  Then $g(d)\mid z_0$, so $d\mid z_0^2$;
and $m\mid 4d+1$ gives $m\mid 4d+4z_0$, so $m\mid d+z_0$.  With
$x=(z_0+d)/m$ and $y=(z_0+z_0^2/d)/m=z_0(d+z_0)/(dm)$, both integers since
$(m,d)=1$, one has $1/x+1/y=m/z_0$, hence
$1/x+1/y+1/(nz_0)=(nm+1)/(nz_0)=4/n$.
\end{proof}

We write \emph{Case-B classes} for the classes of $\cR(M)$ and the
$(a,D)$-classes, and \emph{Case-A classes} for those of
Lemma~\ref{lem:caseA}.  Two facts are used repeatedly.

\begin{lemma}[$n=1$ is never covered; \lab{proved}]\label{lem:one}
No forced class contains $n=1$.
\end{lemma}
\begin{proof}
$4/1>3\geq1/x_1+1/x_2+1/x_3$.
\end{proof}

\begin{lemma}[Mordell's obstruction; classical; {\cite[Lemma~1.1]{TW}}; \lab{proved}]\label{lem:mordell}
Let $M\equiv3\pmod4$, $A=(M+1)/4$ and $D\mid A^2$.  Then $(D,M)=1$ and
the Jacobi symbol satisfies $\bigl(\frac{-4D}{M}\bigr)=-1$.  Consequently,
if $n$ is a nonzero quadratic residue modulo every prime factor of $M$,
then $n$ lies in no class of $\cR(M)$.
\end{lemma}
\begin{proof}
$(A,M)=(A,4A-1)=1$.  Write $D=sr^2$ with $s$ squarefree; then $D\mid A^2$
gives $sr\mid A$, so $M\equiv-1\pmod{4s}$.  Now
$(\frac{-4D}M)=(\frac{-1}M)(\frac sM)=-(\frac sM)$.  If $s$ is odd,
reciprocity and $(\frac Ms)=(\frac{-1}s)$ give $(\frac sM)=1$, since
$(M-1)/2$ is odd.  If $s=2s'$, then $8\mid M+1$, so $(\frac2M)=1$, and
$(\frac{s'}M)=1$ as before.  For the consequence,
$(\frac nM)=\prod_{p^e\| M}(\frac np)^e=1$.
\end{proof}

\section{Congruence sieves: the architecture class}\label{sec:class}

\subsection{CRT systems and majorants}
\begin{definition}[prime-slice CRT system {\cite[\S1]{ET}}]\label{def:slice}
The data are a modulus $Q_0$; a finite set $\cP$ of primes not dividing
$Q_0$ (the \emph{slice primes}); a set $R\subseteq\Z/Q_0$ of admissible
small residues; and, for each $c\in R$ and $\ell\in\cP$, a set
$F_\ell(c)\subseteq\Z/\ell$ of forbidden classes.  The \emph{avoider set} is
\[
 \A=\{n:\ n\bmod Q_0\in R,\ n\bmod\ell\notin F_\ell(n\bmod Q_0)\
 \text{for all }\ell\in\cP\}.
\]
Put $p_\ell(c)=|F_\ell(c)|/\ell$ and $\bar p_\ell=|R|^{-1}\sum_{c\in R}p_\ell(c)$.
\end{definition}
A family of conditions $n\equiv b\pmod{q_0\ell}$ with $q_0\mid Q_0$ and
$\ell\in\cP$ induces such a system with
$F_\ell(c)=\{b\bmod\ell:\ b\equiv c\pmod{q_0}\}$.  A typical $R$ is a
\emph{selector} $R=\{c:(c,P)=1\}$ with $P\mid Q_0$.

\begin{definition}[majorant of level $\lambda$]\label{def:majorant}
A \emph{majorant} of $\A$ is a finite real combination
\[
 \nu(n)=\sum_ia_i\ind[n\equiv b_i\pmod{d_i}]
\]
with $\nu\geq0$ on all of $\Z$ and $\nu\geq1$ on $\A$.  It has
\emph{level} $\lambda$ if $\sum_{\ell\in\cP,\ \ell\mid d_i}\log\ell\leq\lambda$
for every $i$.  Its \emph{mean} is $\E\nu=\sum_ia_i/d_i$, the average
over a common period, and its \emph{saving} is $\log(1/\E\nu)$.
\end{definition}
The level charges only the slice primes; the small part of $d_i$ is free.
This is generous to the sieve.  For every $N$,
\begin{equation}\label{eq:transfer}
 \sum_{n\leq N}\nu(n)=N\cdot\E\nu+O\Bigl(\sum_i|a_i|\Bigr),
\end{equation}
since each class meets $[1,N]$ in $N/d_i+O(1)$ points.

\subsection{The architecture class}
\begin{definition}[CRT-majorant architecture]\label{def:arch}
A \emph{CRT-majorant architecture} for a family $\cF$ of forced classes
bounds $\#(\A\cap[1,N])$, and hence the exceptional integers in
$[1,N]$ that satisfy the admissibility condition $n\bmod Q_0\in R$, as follows.
\begin{enumerate}
\item[(A1)] $\nu$ is a majorant of the avoider set $\A$ of $\cF$ in the
  sense of Definition~\ref{def:majorant}: $\nu\geq0$ on $\Z$ and $\nu\geq1$
  on the \emph{whole} avoider set, not only on exceptional primes.
\item[(A2)] The final bound is $N\cdot\E\nu+T$ with
  $T=\sum_i|a_i|$, and it is non-trivial: $T<N$.
\item[(A3)] The slice primes of the family are at most $N^{A}$ for a fixed
  $A$.
\end{enumerate}
\end{definition}
Every sieve majorant in use satisfies (A1): Bonferroni truncations
$Q_r(H)=\binom{H-1}r$ and Selberg squares.  Both campaign notes satisfy
(A1)--(A3) (Section~\ref{sec:campaign}).

\begin{lemma}[coefficient budget implies level; {\cite[Lemma~2.9]{ET}}, from {\cite[item~3]{ETrev}}; \lab{proved}]\label{lem:budget}
Take a prime-slice system, or the sequential setting of
Section~\ref{sec:seq}.  Let $\nu$ be a majorant of $\A$ with
$T=\sum_i|a_i|$, and assume that every slice prime satisfies
$\log\ell\leq\Lambda_0$.  Then for every $\lambda$ there is a majorant
$\nu'$ of level at most $\lambda$ with $\nu'\geq\nu$ pointwise and
$\E\nu'\leq\E\nu+T\e^{\Lambda_0-\lambda}$.  In particular
$\lambda=\Lambda_0+\log T+\log(1/\E\nu)$ gives $\E\nu'\leq2\E\nu$.
\end{lemma}
\begin{proof}
Keep every term of level at most $\lambda$.  Drop each term of level
$>\lambda$ with $a_i<0$; this raises $\nu$ pointwise and raises the mean
by $|a_i|/d_i\leq|a_i|\e^{-\lambda}$.  For a term of level $>\lambda$ with
$a_i>0$, list its slice primes in increasing order and keep the longest
prefix of level at most $\lambda$.  The next prime has
$\log\ell\leq\Lambda_0$, so the prefix has level $>\lambda-\Lambda_0$.
Replace $d_i$ by the divisor $d_i'$ formed by the non-slice part of $d_i$
and the kept prime powers.  Then
$\ind[n\equiv b_i\ (d_i')]\geq\ind[n\equiv b_i\ (d_i)]$, and the mean rises
by at most $a_i/d_i'\leq a_i\e^{\Lambda_0-\lambda}$.  So $\nu'\geq\nu$,
$\nu'$ is a majorant, and summing gives the bound.
\end{proof}

\begin{corollary}[\lab{proved}]\label{cor:budgetlevel}
Under (A2)--(A3), the majorant $\nu$ can be replaced by a majorant of level
$\lambda\leq(A+1)\log N+s$ and mean at most $2\E\nu$, where
$s=\log(1/\E\nu)$.  Hence, if every majorant of level $\lambda$ has
saving at most $C\lambda^{3/4}+C'$, then $s\ll_{A,C,C'}(\log N)^{3/4}$.
\end{corollary}
\begin{proof}
Lemma~\ref{lem:budget} with $\Lambda_0=A\log N$ and $\log T<\log N$ gives
$\nu'$ of level $\lambda\leq(A+1)\log N+s$ and saving at least $s-\log2$.
So $s\leq\log2+C((A+1)\log N+s)^{3/4}+C'$, which forces
$s\ll_{A,C,C'}(\log N)^{3/4}$.
\end{proof}

\begin{remark}[positivity on all of $\Z$]\label{rem:positivity}
Condition (A1) cannot be weakened to positivity on $[1,N]$.  Restricted
to $[1,N]$, the class $\ind[n\equiv b\pmod N]$ is the point mass at $b$, so
the span of level-$\log N$ class indicators contains every function on
$[1,N]$.  The relaxed linear programme is then the exact count, and
evaluating it presupposes knowing the avoider set
\cite[\S2.4]{ET}.  Such methods are not sieves in the present sense.
\end{remark}

\section{The sieve-limit theorem}\label{sec:limit}

All results of this section are from \cite[\S2]{ET}; the review
\cite[items~1--2]{ETrev} found them sound and checked every intermediate
inequality against exact linear programmes on small systems.  Write
$C_4=8\e^5$.

\subsection{Two one-dimensional lemmas}
\begin{lemma}[binomial mass; \lab{proved}]\label{lem:binom}
Let $\psi=\Bin(z,q)$ with $0<q\leq1/4$ and $\mu'=zq$.  For every integer
$1\leq y\leq z-1$,
$\psi(y)\geq(8y)^{-1/2}\exp\{-\tfrac43(y-\mu')^2/\mu'\}$.
\end{lemma}
\begin{proof}
By \cite[Ch.~10, Lemma~7]{MS},
$\binom zy\geq(8y(1-y/z))^{-1/2}\e^{zH(y/z)}$, so
$\psi(y)\geq(8y)^{-1/2}\e^{-zD(y/z\|q)}$.  The bound
$D(a\|q)\leq(a-q)^2/(q(1-q))$ gives
$zD\leq(y-\mu')^2/(\mu'(1-q))\leq\frac43(y-\mu')^2/\mu'$.
\end{proof}

\begin{lemma}[node lemma; \lab{proved}]\label{lem:node}
Keep $\psi,q,\mu'$ as in Lemma~\ref{lem:binom}.  For a set $Y$ of $k+1$
distinct integers in $[0,z]$ let $B(Y)=\max_{y\in Y}|\ell^Y_y(0)|/\psi(y)$,
where $\ell^Y_y$ is the Lagrange basis polynomial of $Y$.  Then every
polynomial $P$ of degree at most $k$ with $P\geq0$ on $Y$ satisfies
$|P(0)|\leq B(Y)\sum_{y\in Y}\psi(y)P(y)$.  Let $B(k)=\min_YB(Y)$ and
$C_1=16\e^6$.  Then
\begin{enumerate}
\item[(a)] $\log B(0)\leq\frac12\log(8\mu'+8)+2$;
\item[(b)] $\log B(k)\leq4\log(4/3)\mu'\leq1.151\mu'$ for every $k\leq z$;
\item[(c)] if $\mu'\geq64$ and $1\leq k\leq\mu'/16$, then
  $\log B(k)\leq\frac k2\log(C_1\mu'/k)+\frac12\log(16\mu')$.
\end{enumerate}
Consequently, for all $\alpha,s>0$ and $0\leq k\leq z$,
\begin{equation}\label{eq:node}
 \log B(k)\leq19\alpha ks+C_4\mu'\e^{-2\alpha s}+\tfrac12\log(16\mu'+16)+74.
\end{equation}
\end{lemma}
\begin{proof}
The inequality for $P$ is Lagrange interpolation $P(0)=\sum_y\ell_y(0)P(y)$
with $P\geq0$ on $Y$.
(a) If $\mu'\geq1$ then $z\geq4$; take the single node
$y=\lceil\mu'\rceil\leq z-1$ and apply Lemma~\ref{lem:binom} with
$|y-\mu'|\leq1$.  If $\mu'<1$ take the node $0$:
$\psi(0)=(1-q)^z\geq\e^{-1.151\mu'}$.
(b) Take $Y=\{0,\dots,k\}$; then $\ell_y(0)=\delta_{y0}$ and
$B=1/\psi(0)\leq\e^{1.151\mu'}$.
(c) Take nodes $y_i=a+ih$, $0\leq i\leq k$, with
$W=\lceil\sqrt{k\mu'}/2\rceil$, $h=\lceil2W/k\rceil$, $a=\lceil\mu'\rceil-W$.
Since $k\leq\mu'/16$ and $\mu'\geq64$, every node lies in
$[\mu'/2,2\mu']\subseteq[1,z-1]$ and $|y_i-\mu'|\leq\sqrt{k\mu'}$, so
Lemma~\ref{lem:binom} gives $\psi(y_i)\geq(16\mu')^{-1/2}\e^{-4k/3}$.
Next $|\ell_i(0)|=\prod_{j\neq i}y_j/(h^ki!(k-i)!)$, and
$\prod_{j\neq i}y_j\leq(2\mu')^k$, $k!/(i!(k-i)!)\leq2^k$, $k!\geq(k/\e)^k$
and $h\geq\sqrt{\mu'/k}$ give $|\ell_i(0)|\leq(4\e\sqrt{\mu'/k})^k$.
Multiplying gives (c), since $16\e^{2+8/3}\leq C_1$.

For \eqref{eq:node}: when (c) applies, use
$\frac k2\log(C_1\mu'/k)\leq\alpha ks+(C_1\mu'/2\e)\e^{-2\alpha s}$ (the
maximum over real $k$ of the difference), and $C_1/(2\e)=C_4$.  Otherwise
use (a) or (b): if $\mu'<64$ then $1.151\mu'\leq74$; if $k>\mu'/16$ and
$\alpha s\geq1$ then $1.151\mu'\leq18.5k\leq19\alpha ks$; if $\alpha s<1$
then $1.151\mu'\leq8.6\mu'\e^{-2\alpha s}$.
\end{proof}

\begin{lemma}[combination technique; standard; \lab{proved}]\label{lem:comb}
Let $\Lambda\subset\Z_{\geq0}^G$ be a lower set and $P_\Lambda$ the span of
the monomials $x^a$, $a\in\Lambda$.  For each coordinate $g$ and degree $k$
let $I^{(g)}_k$ be Lagrange interpolation at any $k+1$ distinct nodes, and
put $c_j=\sum_{e\in\{0,1\}^G,\ j+e\in\Lambda}(-1)^{|e|}$.  Then for
$Q\in P_\Lambda$,
$Q=\sum_{j\in\Lambda}c_j\bigl(\bigotimes_gI^{(g)}_{j_g}\bigr)Q$, and
$|c_j|\leq2^G$.
\end{lemma}
\begin{proof}
Put $\Delta_k=I_k-I_{k-1}$ with $I_{-1}=0$; expanding
$\sum_{j\in\Lambda}\bigotimes_g\Delta_{j_g}$ gives the coefficients $c_j$.
For $a\in\Lambda$, $\Delta_{j_g}x_g^{a_g}=0$ when $j_g>a_g$, and the box
$\{j\leq a\}$ lies in $\Lambda$, so the sum applied to $x^a$ is
$\prod_g\sum_{j_g\leq a_g}\Delta_{j_g}x_g^{a_g}=x^a$.
\end{proof}

\subsection{The Boolean proposition}
Let $x_i\sim\Bern(p_i)$, $i\in I$, be independent, with weights
$s_i\geq s_*>0$.  A function $f$ on $\{0,1\}^I$ is \emph{$\lambda$-level} if
$f=\sum_Tf_T$ where $f_T$ depends only on $x_T$ and
$\sum_{i\in T}s_i\leq\lambda$.

\begin{proposition}[{\cite[Prop.~2.4]{ET}}; \lab{proved}]\label{prop:boolean}
Assume $p_i\leq1/4$ whenever $s_i\leq\lambda$, and $p_i<1$ for all $i$.  Let
$f$ be $\lambda$-level with $f\geq0$ and $f(0)\geq1$, and let
$\lambda\geq s_*$.  Put $G=\lfloor\log_2(\lambda/s_*)\rfloor+1$ and
$\mu=\sum_{s_i\leq\lambda}p_i$.  Then for every $\alpha>0$,
\begin{equation}\label{eq:boolean}
 \log\frac1{\E f}\leq19\alpha\lambda+C_4\sum_{s_i\leq\lambda}p_i\e^{-\alpha s_i}
 +G\bigl(75+\log(2+\lambda/s_*)\bigr)+\frac G2\log(16\mu+16).
\end{equation}
\end{proposition}
\begin{proof}
\emph{Step 0.} No $f_T$ contains an $i$ with $s_i>\lambda$, so $f$ ignores
those coordinates; discard them, coordinates with $p_i=0$, and empty bands.

\emph{Step 1 (bands).} Put $B_g=\{i:2^gs_*\leq s_i<2^{g+1}s_*\}$ for
$0\leq g<G$, $s_g=2^gs_*$ and $q_g=\max_{i\in B_g}p_i\leq1/4$.

\emph{Step 2 (thinning).} Take independent $u_i\sim\Bern(q_g)$ and
$w_i\sim\Bern(p_i/q_g)$; then $u_iw_i$ has the law of $x_i$.  Fix $w$, let
$Z_g=\{i\in B_g:w_i=1\}$ and $z_g=|Z_g|$.  Then $f_w(u)=f(u\circ w)$
depends only on $u_Z$, is $\lambda$-level and nonnegative, and
$f_w(0)\geq1$.

\emph{Step 3 (symmetrisation).} Average $f_w$ over $\prod_g\mathrm{Sym}(Z_g)$,
which preserves the law of $u$.  Each monomial $u^S$ of a term $f_T$ has
multidegree $(|S\cap Z_g|)_g$ in the lower set
$\Lambda=\{j:\sum_gj_gs_g\leq\lambda\}$, and symmetrises to
$\prod_g\binom{K_g}{|S_g|}/\binom{z_g}{|S_g|}$ with $K_g=|u_{Z_g}|$.  Hence
$\E_uf_w=\E Q(K)$ where $Q\in P_\Lambda$, the $K_g\sim\Bin(z_g,q_g)$ are
independent, $Q\geq0$ on $\prod_g\{0,\dots,z_g\}$ and $Q(0)\geq1$.  Reducing
modulo $\prod_{i=0}^{z_g}(K_g-i)$ we may take $Q\in P_{\Lambda'}$,
$\Lambda'=\Lambda\cap\prod_g[0,z_g]$.

\emph{Step 4 (interpolation).} Choose the node sets of Lemma~\ref{lem:node}
inside $\{0,\dots,z_g\}$ and apply Lemma~\ref{lem:comb} at $0$.  All grid
points are support points, where $Q\geq0$, so
\[
 1\leq Q(0)\leq\Bigl(2^G|\Lambda'|\max_{j\in\Lambda'}\prod_gB_g(j_g)\Bigr)\E Q(K).
\]
(If $z_g=0$ then $j_g=0$ on $\Lambda'$ and $B_g(0)=1$.)

\emph{Step 5 (bounds).} Apply \eqref{eq:node} with $s=s_g$ and
$\mu'=q_gz_g$, using $\sum_gj_gs_g\leq\lambda$ and
$|\Lambda|\leq(1+\lambda/s_*)^G$:
$\log(1/\E_uf_w)\leq\Phi(w)$, where
$\Phi(w)=19\alpha\lambda+\sum_g[C_4q_gz_g\e^{-2\alpha s_g}+\frac12\log(16q_gz_g+16)+74]
+G\log2+G\log(1+\lambda/s_*)$.  By Jensen,
$\E f=\E_w\E_uf_w\geq\e^{-\E_w\Phi(w)}$.  Finally $\E_wq_gz_g=\mu_g:=\sum_{i\in B_g}p_i$,
concavity of $\log$, $\e^{-2\alpha s_g}\leq\e^{-\alpha s_i}$ on $B_g$, and
$\sum_g\log(16\mu_g+16)\leq G\log(16\mu+16)$.
\end{proof}

\subsection{The CRT form}
\begin{theorem}[sieve limit for prime-slice systems; {\cite[Thm~2.5]{ET}}; \lab{proved}]\label{thm:slice}
Take a prime-slice system.  Assume that for every $c\in R$,
$|F_\ell(c)|\leq\ell/4$ for every $\ell\in\cP$ with $\ell\leq\e^\lambda$, and
$|F_\ell(c)|<\ell$ for every $\ell\in\cP$.  Let
$\bar\mu=\sum_{\ell\in\cP,\ \ell\leq\e^\lambda}\bar p_\ell$ be the truncated
mass, $s_*=\log\min\cP$, and $G=\lfloor\log_2(\lambda/s_*)\rfloor+1$.  Then
every majorant $\nu$ of $\A$ of level $\lambda\geq s_*$ satisfies, for every
$\alpha>0$,
\begin{equation}\label{eq:slice}
 \log\frac1{\E\nu}\leq\log\frac{Q_0}{|R|}+19\alpha\lambda
 +C_4\sum_{\ell\in\cP,\ \ell\leq\e^\lambda}\bar p_\ell\,\ell^{-\alpha}
 +G\bigl(75+\log(2+\lambda/s_*)\bigr)+\frac G2\log(16\bar\mu+16).
\end{equation}
If $\lambda<s_*$, $\nu$ depends only on the small residue and the bound
holds without the $\lambda$-terms.
\end{theorem}
\begin{proof}
Use the uniform measure on $\Z/Q_{\rm tot}$ for a common period.  By CRT
the coordinates $n\bmod Q_0^*$ (the $Q_0$-primary part of $Q_{\rm tot}$),
$n\bmod\ell^{e_\ell}$ ($\ell\in\cP$) and the rest are independent and
uniform.  Since $\nu\geq0$,
$\E\nu\geq(|R|/Q_0)\,\mathrm{avg}_{c\in R}\E[\nu\mid n\bmod Q_0=c]$.  Fix
$c\in R$.  The hit indicators $x_\ell=\ind[n\bmod\ell\in F_\ell(c)]$ are
independent $\Bern(p_\ell(c))$, independent of the finer digits.  Each term
of $\nu$, conditioned on $(c,x)$, depends only on $x_{T_i}$ with
$T_i=\{\ell\in\cP:\ell\mid d_i\}$, so $\nu_c=\E[\nu\mid c,x]$ is
$\lambda$-level with weights $s_\ell=\log\ell$, and $\nu_c\geq0$.  The event
$\{c\}\times\{x=0\}$ has positive probability and lies in $\A$, so
$\nu_c(0)\geq1$.  Apply Proposition~\ref{prop:boolean} in each fibre, then
Jensen over $c\in R$; the right side of \eqref{eq:boolean} is linear in
the $p_\ell$ except for the concave logarithmic term.
\end{proof}

\begin{remark}[size of the error terms]\label{rem:errors}
The last two terms of \eqref{eq:slice} are $O(\log^2\lambda)$ when
$s_*\gg1$ and $\log\bar\mu\ll\log\lambda$.  In general $\bar\mu$ may be as
large as $\pi(\e^\lambda)/4$, and then they are $\asymp\lambda\log\lambda$.
In every application below $\bar\mu\ll\lambda^3$.
\end{remark}

\begin{remark}[the $R$-term]\label{rem:Rterm}
The term $\log(Q_0/|R|)$ is a saving that the theorem leaves uncontrolled:
$\nu=\ind_R$ is a majorant with no slice cost and saving exactly
$\log(Q_0/|R|)$.  For a selector $R=\{(c,P)=1\}$ the term is
$\log(P/\varphi(P))\leq\log\log\log P+O(1)$.  If $R$ itself encodes forced
classes with moduli dividing $Q_0$, the term is the void probability of a
non-slice subsystem, about which Theorem~\ref{thm:slice} says nothing.
\end{remark}

Define the \emph{Rankin functional}
\begin{equation}\label{eq:rankin}
 \Psi(\lambda)=\inf_{\alpha>0}\Bigl[\alpha\lambda+\sum_\ell\bar p_\ell\,\ell^{-\alpha}\Bigr].
\end{equation}
Theorem~\ref{thm:slice} says that no majorant of level $\lambda$ saves more
than $C\Psi(\lambda)+O(\log^2\lambda)$ beyond the $R$-term, in the
situation of Remark~\ref{rem:errors}.  No converse is claimed in
general; attainability is discussed only for the two campaign families
(Section~\ref{sec:campaign}).

\begin{remark}[Montgomery's large sieve; {\cite[Remark~2.6]{ET}}; \lab{proved}]\label{rem:LS}
Montgomery's large sieve \cite{Montgomery} applied in a fibre $c$ gives a
bound $Y/S_c(Q)$ with $Q^2\leq Y$, where
$S_c(Q)=\sum_{q\leq Q}\mu^2(q)\prod_{\ell\mid q}g_\ell$ and
$g_\ell=|F_\ell(c)|/(\ell-|F_\ell(c)|)$.  Rankin's trick gives
$S_c(Q)\leq Q^\alpha\prod_\ell(1+g_\ell\ell^{-\alpha})
\leq\exp\{\alpha\log Q+2\sum_\ell p_\ell(c)\ell^{-\alpha}\}$ when
$p_\ell(c)\leq1/2$.  Summed over fibres, Jensen gives
$\sum_cY\e^{-\Phi_c}\geq|R|\,Y\e^{-\mathrm{avg}\,\Phi_c}$.  So the large-sieve
bound over a slice system obeys the same functional, with level
$\lambda=\log Q$, directly and without duality.  No sequential or
shared-prime large-sieve statement is made.
\end{remark}

\subsection{Sequential windows: shared large primes}\label{sec:seq}
Theorem~\ref{thm:slice} needs the slice primes to be disjoint from the
small modulus.  The following extension allows a large prime to be the
slice prime of one condition and a cofactor prime of another.

\emph{Setting} \cite[\S2.6]{ET}.  Take $Q_0$, $R\subseteq\Z/Q_0$, and a set
$\cP$ of primes not dividing $Q_0$, split into ordered windows
$\cP=W_1\sqcup\dots\sqcup W_J$.  A \emph{condition} $C$ is an event
\[
 E_C=\{n\equiv a_C\ (\mathrm{mod}\ q_C),\ n\equiv b_{C,\ell}\
 (\mathrm{mod}\ \ell^{e_{C,\ell}})\ \text{for }\ell\in S(C)\}
\]
with $q_C\mid Q_0$, $\emptyset\neq S(C)\subseteq\cP$ and $e_{C,\ell}\geq1$.
Assume
\begin{itemize}
\item[(U)] the last window meeting $S(C)$ meets it in exactly one prime
  $\ell(C)$, and $e_{C,\ell(C)}=1$.
\end{itemize}
Fix $E_\ell\geq\max_Ce_{C,\ell}$.  The \emph{history} before window $j$ is
$H_{<j}=(n\bmod Q_0,(n\bmod\ell^{E_\ell})_{\ell\in W_{<j}})$.  For
$\ell\in W_j$ and a history $h$ let $F_\ell(h)$ be the set of $b_{C,\ell}$
over the conditions $C$ with $\ell(C)=\ell$ whose other requirements are
met by $h$, and $p_\ell(h)=|F_\ell(h)|/\ell$.  The avoider set is
$\A=\{n:n\bmod Q_0\in R,\ n\notin E_C\ \forall C\}$.  Let $\Qseq$ be the law
of the history built window by window: $c=n\bmod Q_0$ uniform on $R$;
given $H_{<j}$, the residues $n\bmod\ell$ ($\ell\in W_j$) independent and
uniform on $\Z/\ell\setminus F_\ell(H_{<j})$, with uniform independent
higher digits.  Majorants of level $\lambda$ charge all primes of $\cP$.

\begin{theorem}[sequential sieve limit; {\cite[Thm~2.7]{ET}}; \lab{proved}]\label{thm:seq}
Assume that on the support of $\Qseq$, $p_\ell(h)\leq1/4$ whenever
$\log\ell\leq\lambda$, and $p_\ell(h)<1$ always.  Then for all
$\alpha_1,\dots,\alpha_J>0$, every majorant $\nu$ of level $\lambda$ satisfies
\[
 \log\frac1{\E\nu}\leq\log\frac{Q_0}{|R|}+\sum_j\E_{\Qseq}\bigl[\Phi_j(H_{<j})\bigr],
\]
where $\Phi_j(h)$ is the right side of \eqref{eq:boolean} for the
coordinates $\ell\in W_j$ with $\log\ell\leq\lambda$, probabilities
$p_\ell(h)$ and $\alpha=\alpha_j$.  A window with $\log\min W_j>\lambda$
contributes $0$.
\end{theorem}
\begin{proof}
Let $g_j(h)=\E[\nu\mid H_{<j}=h]$ under the uniform measure, and let
$\A_{<j}$ be the histories in $R$ satisfying no condition whose last
window is before $j$.  We show by downward induction that for
$h\in\A_{<j}$,
$-\log g_j(h)\leq\E_{\Qseq}[\sum_{i\geq j}\Phi_i(H_{<i})\mid H_{<j}=h]$.
For $j=J+1$ the history decides membership in $\A$, and $\nu\geq1$ on
$\A$.  For the step, fix $h\in\A_{<j}$.  Given $h$, the residues
$n\bmod\ell^{E_\ell}$, $\ell\in W_j$, are independent and uniform, so the
indicators $x_\ell=\ind[n\bmod\ell\in F_\ell(h)]$ are independent
$\Bern(p_\ell(h))$.  Put $f(x)=\E[g_{j+1}(H_{<j+1})\mid h,x]$.  Then
$\E f=g_j(h)$, $f\geq0$, and $f$ is $\lambda$-level on $W_j$.  The event
$x=0$ is exactly that no condition with last window $j$ holds; given it,
$H_{<j+1}\in\A_{<j+1}$ with the $\Qseq$ transition law.  The induction
hypothesis and Jensen give $f(0)>0$ and the bound for $f(0)$;
Proposition~\ref{prop:boolean} for $f/f(0)$ gives
$\E f\geq f(0)\e^{-\Phi_j(h)}$.  Finally
$\E\nu\geq(|R|/Q_0)\,\mathrm{avg}_{c\in R}g_1(c)$ and Jensen over $c$.
\end{proof}

\begin{lemma}[profile under $\Qseq$; {\cite[Lemma~2.8]{ET}}; \lab{proved}]\label{lem:seqprofile}
For $\ell\in W_j$,
\[
 \E_{\Qseq}p_\ell(H_{<j})\leq\sum_{C:\ \ell(C)=\ell}\frac1\ell\,
 \Prob(c\equiv a_C\ (q_C)\mid c\in R)
 \prod_{\ell'\in S(C)\setminus\{\ell\}}\frac1{\ell'^{e_{C,\ell'}}(1-p^*_{\ell'})},
\]
where $p^*_{\ell'}$ is the supremum of $p_{\ell'}(h)$ over reachable
histories.
\end{lemma}
\begin{proof}
Chain rule over windows: given the past, $n\bmod\ell'$ is uniform on at
least $\ell'(1-p^*_{\ell'})$ residues, and the higher digits are uniform.
\end{proof}

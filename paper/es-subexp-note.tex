\title[A sub-exponential Erd\H{o}s--Straus witness bound]{Large Erd\H{o}s--Straus witnesses II:\\
the multiplier modulus is infinitely often $\exp\big((\log p)^{c}\big)$}
\author{Anonymous}
\date{Draft, 5 October 2026}

\begin{document}

\begin{abstract}
For a prime $p$ let $W(p)$ be the least modulus $M\equiv3\pmod4$ of a
``multiplier'' congruence that yields a solution of $4/p=1/x+1/y+1/z$.  In
earlier work we showed that $W(p)$ exceeds every fixed power of $\log p$ for
infinitely many primes $p$ in Mordell's hard classes.  Here we prove
$W(p)\ge\exp(c(\log p)^{1/14})$ for infinitely many hard primes, assuming
Elsholtz and Tao's average bound for $\tau(kab^2+1)$; equivalently, the least
hard prime with $W(p)>T$ is at most $\exp(O((\log T)^{14}))$.
Unconditionally we get $\log W(p)\ge(1/(2\log2)-o(1))\log\log p\cdot\log\log\log p$
infinitely often.  As before, the class of one modulo the small primes leaves
a residual congruence system on the larger primes.  The local lemma shows
that its avoiding set has Haar density $e^{-O(S)}$, where $S$ is its total
mass.  To transfer this density to primes we need a \emph{pointwise}
minorant that is a short combination of congruence classes, each on few
primes.  We take the one-sided $\ell^2$ sandwich of Bazzi and Razborov from
the theory of bounded independence.  Its error is the Efron--Stein tail of
the avoidance indicator.  After a binary encoding of residues the indicator
is a CNF of small width, so Håstad's switching lemma, through the argument
of Linial, Mansour and Nisan, bounds the tail.  The minorant is transferred
to primes by the Linnik-range prime number theorem of Thorner and Zaman.  No
codegree condition, level structure or truncated inclusion--exclusion is
needed.
\end{abstract}

\section{Introduction}

The Erd\H{o}s--Straus conjecture asserts that $4/n=1/x+1/y+1/z$ is solvable
in positive integers for every $n\ge2$.  It suffices to treat primes.  It has
been verified for $n\le10^{18}$ (\cite{MD}; \cite{Salez} for $10^{17}$), and
Mordell's identities settle every prime outside the six classes
$p\equiv1^2,11^2,13^2,17^2,19^2,23^2\pmod{840}$ \cite{Mordell} (cf.\
\cite{Oblath}).  We call the primes in these classes \emph{hard}.  For the
exceptional set see Vaughan \cite{Vaughan} and \cite{PW}; for the average
number of solutions see \cite{ET}.

For $M\equiv3\pmod4$ put $A_M=(M+1)/4$.  If $uvw=A_M$ and
$n\equiv-uv^{-1}\pmod M$, then $s=(nv+u)/M$ is a positive integer and
\begin{equation}\label{eq:ident}
 \frac4n=\frac1{suw}+\frac1{nsvw}+\frac1{nuvw}
\end{equation}
(direct verification; this is the classical multiplier family, cf.\
\cite{Vaughan,ET}).  Define
\begin{equation}\label{eq:W}
 W(n)=\min\{M\equiv3\ (4):\ n\equiv-uv^{-1}\ (\mathrm{mod}\ M)\ \text{for some}\ uvw=A_M\},
\end{equation}
with $W(n)=+\infty$ if no such $M$ exists.  Thus $W(p)\le T$ provides an
explicit representation with modulus at most $T$.  Every prime
$p\equiv1\pmod{\lcm\{M\le T\}}$ has $W(p)>T$ (Lemma~\ref{lem:one}), so
Linnik's theorem gives $W(p)\gg\log p$ infinitely often.  In
\cite{Paper1} we proved that for every fixed $k$,
$W(p)\ge(\log p)^{k}e^{-C_k\log\log p/\log\log\log p}$ for infinitely many
hard primes.  Here we prove a sub-exponential rate.

\begin{theorem}[Proved modulo Theorem~\ref{thm:TZ} and {\cite[Prop.~1.4]{ET}}]\label{thm:main}
There is an absolute constant $c>0$ such that for infinitely many hard primes $p$
\[
 W(p)\ \ge\ \exp\big(c\,(\log p)^{1/14}\big).
\]
More precisely, for every sufficiently large $T$ there is a prime
$p\equiv1\pmod{840}$ with $p>T$, $W(p)>T$ and $\log p\le C(\log T)^{14}$.
\end{theorem}

\begin{theorem}[Proved modulo Theorem~\ref{thm:TZ}]\label{thm:uncond}
For infinitely many hard primes $p$,
\[
 \log W(p)\ \ge\ \Big(\frac1{2\log2}-o(1)\Big)\log\log p\cdot\log\log\log p .
\]
\end{theorem}

Theorem~\ref{thm:TZ} is Thorner and Zaman's prime number theorem in the
Linnik range \cite{TZ}.  No hypothesis on Siegel zeros is needed.
\cite[Prop.~1.4]{ET} is used only to bound the total mass of the residual
system by $(\log T)^4\log\log T$ (Lemma~\ref{lem:qmass}).  Without it,
the same argument gives Theorem~\ref{thm:uncond}.  All constants are
effective, except that we have not checked whether the implied constant in
\cite[Prop.~1.4]{ET} is.  We make no numerical claim.

\subsection*{The method}
We fix $T$ and impose $n\equiv1$ modulo the prime powers $\le T$ of a set
$\Pi$ of primes, consisting of the primes up to $z=(\log T)^2$ and a few
further ones.  Then $W(n)>T$ whenever $n$ avoids a finite family of
congruence \emph{events}, each on at most $k\approx\log T/(2\log\log T)$ of
the remaining (\emph{free}) primes (\S\ref{sec:system}).  In the Haar
measure on residues the local lemma shows that the avoiding set has density
$\delta\ge e^{-2.2S}$, where $S\ll(\log T)^4\log\log T$ is the total
mass, once an iterated quarantine has made the mass at each free prime
small.  To find a \emph{prime} in the avoiding set we need a function
$B=\sum_ic_i\mathbf 1[n\equiv b_i\ (d_i)]$ with $B\le\mathbf 1[\text{avoid}]$
pointwise, mean $\gg\delta$, controlled $\ell^1$ mass, and moduli $d_i$ that
each involve few primes.  The transfer theorem (Theorem~\ref{thm:transfer},
from \cite{Paper1}) then gives a prime $p$ with
$\log p\ll K\max(\log Z,K)$, where $K=1+\log(\ell^1/\text{mean})$ and $Z$ is
the largest modulus.  Both are polynomial in $\log T$ in our construction.

In \cite{Paper1} the minorant was an alternating expansion (inclusion--exclusion
truncated by the number of primes in the support), organised in levels.
That route loses a factorial in the number of levels, which limits it to
fixed powers of $\log p$.  Here we use instead the one-sided
$\ell^2$-approximation scheme of Bazzi \cite{Bazzi}, in the form of
Razborov \cite{Razborov} (Lemma~\ref{lem:brw}).  For events
$E_1,\dots,E_m$ with indicators $A_i$ and arbitrary functions $u_j$,
\[
 B=1-\sum_iA_i\Big(1-\sum_{j<i}A_ju_j\Big)^2\ \le\ \mathbf 1[\text{no }E_i\text{ occurs}],
\]
and $\E[F-B]\le m^2\sum_j\PP(E_j)\,\E(F^{(j)}-u_j)^2$, where $F^{(j)}$ is the
indicator that no earlier event occurs, conditioned on $E_j$.  Choosing $u_j$
as the best $\ell^2$ approximation of $F^{(j)}$ by sums of functions of at
most $t$ coordinates makes every cell of $B$ involve at most $3k+2t$ primes,
and makes the error an \emph{Efron--Stein tail} of $F^{(j)}$.  We bound this
tail in \S\ref{sec:tail}.  Encode each residue in binary.  Then
$1-F^{(j)}$ becomes a DNF of width $k\log_2(4T^2)$, and Håstad's switching
lemma with the Linial--Mansour--Nisan argument \cite{Hastad,LMN} gives
tail $\le4\cdot2^{-k_0}$ at level $t\asymp k\log T\cdot k_0$
(Lemma~\ref{lem:tail}).  We need $k_0\asymp S+k\log T$, so
$t\ll(\log T)^6$ and $\log p\ll(\log T)^{14}$.

\subsection*{Relation to the literature}
Lemma~\ref{lem:brw} is Razborov's identity, and Lemma~\ref{lem:tail} is a
routine application of \cite{Hastad,LMN} (we follow \cite[\S4]{OD}).  We claim
no novelty for either.  To our knowledge these tools from the theory of
bounded independence (``polylogarithmic independence fools DNF and
$\mathrm{AC}^0$'' \cite{Bazzi,Razborov,Braverman}) have not been used
before to construct sieve minorants for primes.  The point is that, in the
Linnik range, primes $p\equiv1\pmod Q$ distribute over congruence cells
with a \emph{relative} error.  So the number of primes per cell is what
costs, while the size of the moduli is nearly free.  Earlier uses of the
switching lemma in number theory known to us run in the opposite
direction: they show that an arithmetic function such as the M\"obius
function is uncorrelated with bounded-depth circuits.  Our literature
search was limited, and we make no priority claim.  The Haar-side input,
a local-lemma lower bound for the density of the uncovered set of a
congruence system, has well-known relatives in the work on Erd\H{o}s
covering systems by Hough and by Balister, Bollob\'as, Morris,
Sahasrabudhe and Tiba.  We do not use those results.

\section{Atoms and the residual system}\label{sec:system}

For $M\equiv3\pmod4$ put $\cR(M)=\{-4D\bmod M:\ D\mid A_M^2\}$.  Note that
$\gcd(A_M,M)=1$ (as $4A_M=M+1$), so every element of $\cR(M)$ is a unit
class.

\begin{lemma}[Proved; cf.\ {\cite{Paper1}}]\label{lem:atoms}
$\{-uv^{-1}\bmod M:\ uvw=A_M\}=\cR(M)$.  Consequently, for every integer $n$,
$W(n)>T$ if and only if $n\bmod M\notin\cR(M)$ for every $M\le T$ with
$M\equiv3\pmod4$.
\end{lemma}

\begin{proof}
Since $4A_M\equiv1\pmod M$, from $uvw=A_M$ we get $v^{-1}\equiv4uw$, hence
$-uv^{-1}\equiv-4u^2w$ and $D=u^2w$ divides $A_M^2$.  Conversely let
$D\mid A_M^2$.  For each prime $q$ with $q^a\Vert A_M$ and $q^d\Vert D$
($d\le2a$) put $u_q=\max(0,d-a)$, $w_q=d-2u_q$, $v_q=a-u_q-w_q$; these are
non-negative, so $u=\prod q^{u_q}$, $v=\prod q^{v_q}$, $w=\prod q^{w_q}$
satisfy $uvw=A_M$ and $u^2w=D$.
\end{proof}

\begin{lemma}[Class of one; proved]\label{lem:one}
$1\notin\cR(M)$ for every $M\equiv3\pmod4$.
\end{lemma}

\begin{proof}
If $-4D\equiv1$ then $D\equiv-A_M\pmod M$ (multiply by $A_M$).  Since
$4A_M^2\equiv A_M$, the map $D\mapsto A_M^2/D$ preserves this congruence, so
we may assume $D\le A_M$; then $0<D+A_M\le2A_M<M$, while $M\mid D+A_M$.
\end{proof}

An \emph{atom} is a pair $(M,D)$ with $M\le T$, $M\equiv3\pmod4$ and
$D\mid A_M^2$.  For a set $\Pi$ of primes write $M=m_\Pi r_\Pi$, where
$m_\Pi$ is the largest divisor of $M$ composed of primes in $\Pi$.  The atom
\emph{survives} $\Pi$ if $m_\Pi\mid4D+1$ and $r_\Pi>1$; it then defines the
\emph{event} $n\equiv-4D\pmod{r_\Pi}$, a unit class, of Haar weight
$1/\varphi(r_\Pi)$.  Put
\[
 S^*=S^*(T)=\sum_{(M,D)}\ \max_{\Pi}\ \frac{\mathbf 1[(M,D)\text{ survives }\Pi]}{\varphi(r_\Pi)},
\]
the sum over all atoms and the maximum over all sets of primes.

\begin{lemma}[Mass bound; proved, and (b) modulo {\cite[Prop.~1.4]{ET}}]\label{lem:qmass}
\textup{(a)} $S^*\le\exp\big((\log2+o(1))\Lc/\log\Lc\big)$.
\textup{(b)} $S^*\ll\Lc^4\log\Lc$.
\end{lemma}

\begin{proof}
Let $(M,D)$ survive $\Pi$, put $A=A_M$ and $g=\gcd(M,4D+1)$.  Then
$m_\Pi\mid g$, and $\varphi(M)\le m_\Pi\varphi(r_\Pi)$, so
$1/\varphi(r_\Pi)\le g/\varphi(M)\le C\,g\log\Lc/M$ for an absolute $C$.  This
bound does not depend on $\Pi$, so
$S^*\le C\log\Lc\sum_{(M,D)}g/M$.

The map $D\mapsto D'=A^2/D$ preserves $g$.  Indeed, $D$ is a unit modulo $M$,
and for a prime power $q^a\mid M$ with $q^a\mid4D+1$ we have
$4D'\equiv4A^2\cdot(-4)\equiv-(4A)^2\equiv-1\pmod{q^a}$.  It exchanges
$D<A$ with $D>A$, so it suffices to sum over $D\le A$ and double.  Write
$D=sr^2$ with $s$ squarefree.  Then $D\mid A^2$ if and only if $sr\mid A$;
write $A=srk$, so that $D\le A$ if and only if $r\le k$, and
$M=4srk-1\ge3srk$.  From $g\mid4sr^2+1$ and $g\mid M=4srk-1$ we get
$g\mid4sr(r+k)$, and $\gcd(g,4sr)=1$, so $g\mid\gcd(4sr^2+1,\,r+k)$.  For fixed
$(s,r)$ the triple $(M,D,g)$ is determined by $k$.  Using
$g\le\sum_{e\mid g}e$, we get
\[
 \sum_{\substack{r\le k\le X}}\frac{g}{k}\ \le\ \sum_{e\mid4sr^2+1}e\sum_{\substack{r\le k\le X\\ k\equiv-r\ (e)}}\frac1k
 \ \le\ (3+\log X)\,\tau(4sr^2+1),\qquad X=\tfrac{T+1}4 .
\]
For the last step: the least admissible $k$ is at least $\max(r,e-r)\ge e/2$,
so the inner sum is at most $2/e+(1+\log X)/e$.  Hence
\[
 S^*\ \le\ C'\log\Lc\cdot\Lc\sum_{sr^2\le X}\frac{\tau(4sr^2+1)}{sr}.
\]
(a) Bound $\tau(4sr^2+1)\le\tau^*(T+2)$ and $\sum_{s,r\le X}1/(sr)\le(1+\Lc)^2$.
Then use $\log\tau^*(x)\le(\log2+o(1))\log x/\log\log x$ \cite[Thm.~317]{HW}.
(b) Split $s\in[2^i,2^{i+1})$, $r\in[2^j,2^{j+1})$ with $2^i,2^j\le X$.  By
\cite[Prop.~1.4]{ET} with $k=4$, $A=2^{i+1}$, $B=2^{j+1}$ (so $A,B>1$ and
$k\ll(AB)^{O(1)}$), each block contributes
$\ll2^{-i-j}AB\log(A+B)\ll\Lc$.  There are $\ll\Lc^2$ blocks, so the sum is
$\ll\Lc^3$.
\end{proof}

\begin{lemma}[Iterated quarantine; proved, cf.\ {\cite{Paper1}}]\label{lem:iterq}
Let $2\le z\le T$, $k=\lfloor\Lc/\log z\rfloor$ and $0<c_0\le1$.  For a set
$\Pi\supseteq\{\ell\le z\}$ and a prime $\ell\notin\Pi$ let $w_\ell(\Pi)$ be
the total Haar weight of the distinct events of atoms surviving $\Pi$ whose
modulus $r_\Pi$ is divisible by $\ell$.  Start from $\Pi_0=\{\ell\le z\}$ and
repeatedly add to $\Pi$ every prime $\ell\notin\Pi$ with $w_\ell(\Pi)>c_0$.
The process stops at some $\Pi=\Pi_0\cup\cB$ with $|\cB|\le kS^*/c_0$, and then
$w_\ell(\Pi)\le c_0$ for every prime $\ell\notin\Pi$.
\end{lemma}

\begin{proof}
The process stops since only primes $\le T$ can have $w_\ell>0$.  If $\ell$ is
added at a stage $\Pi_i$, then $c_0<w_\ell(\Pi_i)$, which is at most the sum
of the contributions to $S^*$ of the atoms $(M,D)$ with $\ell\mid M$.  Summing
over the added primes, each atom is counted at most once for each prime
factor $\ell>z$ of $M$.  There are at most $k$ such factors since $M\le T$.
\end{proof}

\begin{definition}[Residual system]\label{def:system}
Let $\Pi\supseteq\{\ell\le z\}$ be as in Lemma~\ref{lem:iterq}, let
$e_\ell\ge1$ be maximal with $\ell^{e_\ell}\le T$, and put
$Q_\Pi=\lcm(24,\ \ell^{e_\ell}:\ \ell\in\Pi,\ \ell\le T)$.  The \emph{free}
primes are $\cP=\{\ell\le T:\ \ell\notin\Pi\}$, with \emph{coordinates}
$X_\ell=n\bmod\ell^{e_\ell}$ taking values in $G_\ell=(\mathbb Z/\ell^{e_\ell})^\times$.
The \emph{Haar measure} $\PP$ makes the $X_\ell$ independent and uniform.
For each \emph{distinct} pair $(r,c)=(r_\Pi,\,-4D\bmod r_\Pi)$ arising
from an atom $(M,D)$ surviving $\Pi$, with $r=\prod\ell^{a_\ell}$, and each
\emph{lift}, i.e.\ each $\sigma\in\prod_{\ell\mid r}G_\ell$ with
$\sigma_\ell\equiv c\pmod{\ell^{a_\ell}}$, we take the event
$E=\{X_\ell=\sigma_\ell\ \text{for all}\ \ell\mid r_\Pi\}$, with support
$\supp E=\{\ell\mid r\}$.  Let $E_1,\dots,E_m$ list these events (two
pairs may have a common lift; such duplicates are kept), and let $F=\mathbf 1[\text{no }E_i\text{ occurs}]$.
\end{definition}

The lifts of one pair $(r,c)$ partition the class $c\bmod r$, so their weights
add up to $1/\varphi(r)$.

\begin{lemma}[Properties of the residual system; proved]\label{lem:system}
In Definition~\ref{def:system}, with $z\ge7$:
\begin{enumerate}
\item[(i)] $840\mid Q_\Pi$ and $\log Q_\Pi\le(\pi(z)+|\cB|)\Lc+\log24$;
\item[(ii)] $|\cP|\le T$, $|G_\ell|\le T$, $|\supp E_i|\le k$, and $m\le T^{k+2}$;
\item[(iii)] $w_\ell:=\sum_{i:\,\ell\in\supp E_i}\PP(E_i)=w_\ell(\Pi)$ for every
$\ell\in\cP$, and $S:=\sum_{i\le m}\PP(E_i)\le S^*$; so if $\Pi$ is the
output of Lemma~\ref{lem:iterq}, then $w_\ell\le c_0$ for all $\ell\in\cP$;
\item[(iv)] \textup{(Reduction.)} If $n\equiv1\pmod{Q_\Pi}$ and $F(n)=1$
(the residues $X_\ell=n\bmod\ell^{e_\ell}$ need not be units), then $W(n)>T$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) $24\mid Q_\Pi$ and $5,7\in\Pi$.  (ii) An event's support consists of the
prime factors $>z$ of some $M\le T$, so there are at most $k$ of them.  There
are at most $T^2$ atoms, since $\tau(A_M^2)\le2A_M\le T$, and each has at most
$\prod_{\ell\mid r_\Pi}\ell^{e_\ell-a_\ell}\le T^k$ lifts.  (iii) Lifting
preserves weights.  Each pair $(r,c)$ comes from an atom whose contribution to
$S^*$ is at least $1/\varphi(r)$.  (iv) Let $M\le T$ with $M\equiv3\pmod4$, and suppose
$n\equiv-4D\pmod M$ with $D\mid A_M^2$.  Every $\ell^v\Vert M$ with $\ell\in\Pi$
has $\ell^v\le T$, so $m_\Pi\mid Q_\Pi$ and $-4D\equiv n\equiv1\pmod{m_\Pi}$,
i.e.\ $m_\Pi\mid4D+1$.  If $r_\Pi=1$ this contradicts Lemma~\ref{lem:one}.
Otherwise the atom survives $\Pi$, and $n\bmod\ell^{e_\ell}$ (a unit, as
$-4D$ is) is a lift of $-4D\bmod\ell^{a_\ell}$ for each $\ell\mid r_\Pi$.  So
a lift of the pair $(r_\Pi,-4D\bmod r_\Pi)$ occurs, contradicting $F(n)=1$.  Conclude with
Lemma~\ref{lem:atoms}.
\end{proof}

\section{The analytic input and the transfer theorem}\label{sec:analytic}

\begin{theorem}[Thorner--Zaman {\cite[Cor.~1.4 and Rem.~1.5]{TZ}}; cited]\label{thm:TZ}
There are absolute effectively computable constants $c_4>0$, $C_0$ such
that for coprime $a,q$ with $q\ge2$ and $x\ge q^{12}$,
\begin{gather*}
 \sum_{\substack{p\le x\\ p\equiv a\,(q)}}\log p=\frac{\lambda x}{\varphi(q)}\,(1+\varepsilon),\\
 |\varepsilon|\le C_0\Big(e^{-c_4\log x/\log q}+e^{-c_4(\log x)^{3/5}/(\log\log x)^{1/5}}\Big),
\end{gather*}
where $\lambda=1-\chi_1(a)x^{\beta_1-1}/\beta_1$ if the exceptional zero $\beta_1$
for the modulus $q$ exists and $\lambda=1$ otherwise.  Here, following
McCurley \cite{McC} as quoted in \cite[p.~1]{TZ}, $\prod_{\chi\bmod q}L(s,\chi)$
has no zero in $\Re s\ge1-1/(13\log(q(|\Im s|+3)))$ apart from at most one
real zero $\beta_1$, which is simple and belongs to a unique real character
$\chi_1$; also $0<\lambda<2$.
\end{theorem}

We used the arXiv version; we did not compare the numbering with the
published version.  We also use the quoted McCurley statement for moduli
other than those of Theorem~\ref{thm:TZ}; alternatively the classical
Landau--Page theorem \cite[Ch.~14]{Dav} with a constant $c\le1/13$ in place
of $1/13$ works verbatim.

\begin{lemma}[One severe character per family; proved modulo the McCurley statement]\label{lem:severe}
Let $\cQ$ be a finite set of moduli, all $\le Z$.  Call $(\chi^*,\beta)$
\emph{severe} if $\chi^*$ is a real primitive character whose conductor $q^*$
divides some $q\in\cQ$, $L(\beta,\chi^*)=0$ and $\beta\ge1-1/(13\log(3Z^2))$.
Then there is at most one severe pair.  If $(\chi^*,\beta^*)$ is severe and
$q^*\mid q\in\cQ$, then $\beta_1=\beta^*$ and $\chi_1$ is induced by $\chi^*$ for
the modulus $q$.  If $q\in\cQ$ and no severe pair has $q^*\mid q$, then any exceptional
$\beta_1$ of $q$ satisfies $x^{\beta_1-1}/\beta_1\le2\exp(-\log x/(13\log3Z^2))$.
\end{lemma}

\begin{proof}
If $(\chi_a,\beta_a),(\chi_b,\beta_b)$ are severe with conductors dividing
$q_a,q_b\in\cQ$, put $q'=\lcm(q_a,q_b)\le Z^2$.  Both characters induce
characters mod $q'$ with the same zeros in $\Re s>0$, and both $\beta$ lie in
McCurley's region for $q'$; uniqueness there gives $\beta_a=\beta_b$ and
$\chi_a=\chi_b$.  The second claim holds because $\beta^*$ lies in McCurley's
region for $q\le Z$.  For the third, an exceptional zero of $q$ in
$[1-1/(13\log 3Z^2),1)$ would be severe; also $\beta_1\ge1-1/(13\log6)>1/2$.
\end{proof}

\section{A transfer theorem}\label{sec:transfer}

\begin{theorem}[Proved modulo Theorem~\ref{thm:TZ}; from {\cite{Paper1}}]\label{thm:transfer}
Let $T\ge1$, $Q\ge2$, and let $B(n)=\sum_{i\in I}c_i\,\mathbf 1[n\equiv b_i\ (d_i)]$
be a finite real combination with $\gcd(d_i,Q)=\gcd(b_i,d_i)=1$ and
$B(n)\le\mathbf 1[W(n)>T]$ for all $n\equiv1\pmod Q$.  Put
$Z=Q\max d_i$, $\mu=\sum c_i/\varphi(d_i)$, $M_1=\sum|c_i|/\varphi(d_i)$, and assume
$\mu>0$ and the \emph{twist condition}: for every real primitive character
$\psi$ of conductor $f>1$ with $\gcd(f,Q)=1$ and $f\mid d_i$ for some $i$,
\[
 |\mu_\psi|\le\mu/4,\qquad \mu_\psi=\sum_{i:\,f\mid d_i}c_i\psi(b_i)/\varphi(d_i).
\]
Let $K=1+\log(M_1/\mu)$.  There is an absolute $C_1$ such that if
$\log x\ge C_1K\max(\log Z,K)$, then some prime $p\le x$, $p\equiv1\pmod Q$,
has $W(p)>T$.
\end{theorem}

\begin{proof}
Let $q_i=Qd_i$ and let $a_i$ be the class $\equiv1\ (Q)$, $\equiv b_i\ (d_i)$;
it is a unit class.  Then
$\sum_{p\le x,\,p\equiv1(Q)}\log p\,B(p)=\sum_ic_i\theta(x;q_i,a_i)$.  Since
$M_1\ge\mu$ we have $K\ge1$, so taking $C_1\ge12$ gives
$\log x\ge12\log Z$, i.e.\ $x\ge Z^{12}\ge q_i^{12}$, and
Theorem~\ref{thm:TZ} applies to every term.  Put
$E_1=e^{-c_4\log x/\log Z}+e^{-c_4(\log x)^{3/5}/(\log\log x)^{1/5}}$ and
$E_2=e^{-\log x/(13\log3Z^2)}$.  Since $\gcd(Q,d_i)=1$ we have
$\varphi(q_i)=\varphi(Q)\varphi(d_i)$, and $q_i\le Z$, so
\[
 \sum_ic_i\theta(x;q_i,a_i)=\frac{x}{\varphi(Q)}\sum_i\frac{c_i\lambda_i(1+\varepsilon_i)}{\varphi(d_i)},\qquad
 |\varepsilon_i|\le C_0E_1,
\]
with $\lambda_i$ the factor of Theorem~\ref{thm:TZ} for $(q_i,a_i)$.  Apply
Lemma~\ref{lem:severe} to $\cQ=\{q_i\}$; a severe $\beta$ satisfies
$\beta\ge1-1/(13\log 6)>1/2$.  Write the conductor of a severe $\chi^*$ as $q^*=q^*_Qq^{*\prime}$ with
$q^*_Q$ supported on the primes of $Q$ and $\gcd(q^{*\prime},Q)=1$, and
$\chi^*=\chi^*_Q\chi^{*\prime}$.

\emph{Case A: a severe pair exists with $q^*\mid Q$.}  Then $q^*\mid q_i$ for
every $i$.  By the second claim of Lemma~\ref{lem:severe}, every $\lambda_i$ is
formed with the same pair $(\chi^*,\beta^*)$, so
$\lambda_i=1-\chi^*(a_i)x^{\beta^*-1}/\beta^*=1-x^{\beta^*-1}/\beta^*=:\lambda_*$,
because $a_i\equiv1\pmod{q^*}$.  Moreover $\lambda_*>0$: since $\log x\ge12\log Z>2$ and
$1/2<\beta^*<1$,
$\log\beta^*\ge1-1/\beta^*>2(\beta^*-1)>(\beta^*-1)\log x$, i.e.\
$x^{\beta^*-1}/\beta^*<1$.
So
$\sum c_i\theta\ge(x\lambda_*/\varphi(Q))(\mu-C_0E_1M_1)$.

\emph{Case B: otherwise.}  If a severe $\chi^*$ exists, then $q^*$ divides some
$Qd_j$ with $\gcd(d_j,Q)=1$, so $q^*_Q\mid Q$.  Hence $q^*\mid q_i$ iff
$q^{*\prime}\mid d_i$, and in Case~B we have $q^{*\prime}>1$.  For such $i$,
$\chi_1=\chi^*$ on units mod $q_i$ (Lemma~\ref{lem:severe}), and, as
$a_i\equiv1\pmod{q^*_Q}$ and $a_i\equiv b_i\pmod{q^{*\prime}}$,
$\chi^*(a_i)=\chi^*_Q(1)\chi^{*\prime}(b_i)=\psi(b_i)$ with $\psi=\chi^{*\prime}$.  So
$\lambda_i=1-\psi(b_i)x^{\beta^*-1}/\beta^*$.  This $\psi$ is real primitive (the
coprime-conductor factors of a real primitive character are real primitive), of
conductor $q^{*\prime}>1$ coprime to $Q$ and dividing some $d_i$, so the twist
condition applies to it.  For all other $i$ (and for all $i$ if there is no
severe pair), $|\lambda_i-1|\le x^{\beta_1-1}/\beta_1\le2E_2$ by the third claim of
Lemma~\ref{lem:severe}.  Since $\lambda_i<2$ and
$x^{\beta^*-1}/\beta^*\le1$, splitting $\lambda_i=1+(\lambda_i-1)$ gives
\begin{align*}
 \sum c_i\theta&\ge\frac{x}{\varphi(Q)}\big(\mu-2|\mu_\psi|-M_1(2C_0E_1+2E_2)\big)\\
 &\ge\frac{x}{\varphi(Q)}\big(\mu/2-M_1(2C_0E_1+2E_2)\big),
\end{align*}
using $|\mu_\psi|\le\mu/4$.  Both brackets are positive once
$E_1,E_2\le\mu/(8(C_0+1)M_1)=e^{1-K}/(8(C_0+1))$.  This holds if $C_1$ is large:
$\log x/\log Z\ge C_1K$, $(\log x)^{3/5}/(\log\log x)^{1/5}\ge(\log x)^{1/2}\ge
C_1^{1/2}K$ and $\log x/(13\log3Z^2)\ge C_1K/52$ (as $Z\ge2$), so each of the
three exponentials is at most $e^{-cC_1^{1/2}K}$ with $c>0$ absolute.  Hence
$\sum_p\log p\,\mathbf 1[W(p)>T]\ge\sum_p\log p\,B(p)>0$.
\end{proof}

No lower bound for $1-\beta^*$ is used: in Case~A the exceptional factor
$\lambda_*$ multiplies main term and error alike.  The theorem and its proof
are taken verbatim from \cite{Paper1}.

\begin{remark}[Assessment]\label{rem:linnik}
In \cite{Paper1} the Linnik-range uniformity was essential, because there
$K\le T^{o(1)}$ while $\log Z\ge T^{1/k}$.  In the present application $K$
and $\log Z$ are both of order $(\log T)^7$
(\S\ref{sec:proof}).  So a classical prime number theorem of Page type, with error
$O(xe^{-c\sqrt{\log x}})$ uniformly for $q\le e^{c\sqrt{\log x}}$ and with
the exceptional zero treated as above, would require
$\log x\gg(\log Z+K)^2$.  That is the same order
$(\log T)^{14}$.  We have not written out this variant and do not
use it.
\end{remark}
